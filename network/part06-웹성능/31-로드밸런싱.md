# Chapter 31. 로드밸런싱

## 31.1 로드밸런싱이란?

### 정의

```
Load Balancing:
트래픽을 여러 서버에 분산

비유: 마트 계산대
┌────────────────────────────────┐
│ 1개 계산대 → 줄 길어짐 (느림) │
│ 5개 계산대 → 줄 짧아짐 (빠름) │
│ 손님 자동 배분 = 로드밸런서    │
└────────────────────────────────┘
```

### 필요성

```
단일 서버 한계:
┌──────────────────┐
│ 요청: 10,000 RPS │
│ 서버 용량: 1,000 │
│ → 🔥 서버 다운   │
└──────────────────┘

로드밸런싱:
┌──────────────────────────┐
│ 요청: 10,000 RPS         │
│ → 서버 10대              │
│ → 각 1,000 RPS           │
│ ✅ 안정적                │
└──────────────────────────┘
```

## 31.2 로드밸런싱 알고리즘

### Round Robin

```
순서대로 배분

서버: A, B, C

요청 1 → A
요청 2 → B
요청 3 → C
요청 4 → A
요청 5 → B
...

장점:
✅ 간단함
✅ 균등 분배

단점:
❌ 서버 성능 차이 무시
❌ 세션 유지 어려움
```

### Weighted Round Robin

```
가중치 기반 배분

서버 A: 가중치 3
서버 B: 가중치 2
서버 C: 가중치 1

요청 1 → A
요청 2 → A
요청 3 → A
요청 4 → B
요청 5 → B
요청 6 → C

사용 예:
- 서버 A: 고성능 (3배)
- 서버 B: 중간 (2배)
- 서버 C: 저성능 (1배)
```

### Least Connections

```
연결 수가 가장 적은 서버에 배분

현재 상태:
서버 A: 10 connections
서버 B: 5 connections
서버 C: 8 connections

새 요청 → B (가장 적음)

장점:
✅ 부하 고려
✅ 긴 연결에 유리

사용 예:
- 데이터베이스 연결
- WebSocket
```

### IP Hash

```
클라이언트 IP 기반 할당

hash(클라이언트 IP) % 서버 수

예:
hash(192.168.1.100) % 3 = 1 → 서버 B
hash(192.168.1.101) % 3 = 2 → 서버 C
hash(192.168.1.102) % 3 = 0 → 서버 A

장점:
✅ 같은 IP는 같은 서버
✅ 세션 유지

단점:
❌ IP 변경 시 다른 서버
❌ 불균등 분배 가능
```

## 31.3 Nginx 로드밸런싱

### 기본 설정

```nginx
# nginx.conf

http {
    # Upstream 정의
    upstream backend {
        # Round Robin (기본)
        server 192.168.1.101:3000;
        server 192.168.1.102:3000;
        server 192.168.1.103:3000;
    }

    server {
        listen 80;
        server_name example.com;

        location / {
            proxy_pass http://backend;

            # 프록시 헤더
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }
    }
}
```

### Weighted Round Robin

```nginx
upstream backend {
    server 192.168.1.101:3000 weight=3;  # 3배
    server 192.168.1.102:3000 weight=2;  # 2배
    server 192.168.1.103:3000 weight=1;  # 1배
}
```

### Least Connections

```nginx
upstream backend {
    least_conn;  # Least Connections 알고리즘

    server 192.168.1.101:3000;
    server 192.168.1.102:3000;
    server 192.168.1.103:3000;
}
```

### IP Hash

```nginx
upstream backend {
    ip_hash;  # IP 기반 해시

    server 192.168.1.101:3000;
    server 192.168.1.102:3000;
    server 192.168.1.103:3000;
}
```

### Health Check

```nginx
upstream backend {
    server 192.168.1.101:3000 max_fails=3 fail_timeout=30s;
    server 192.168.1.102:3000 max_fails=3 fail_timeout=30s;
    server 192.168.1.103:3000 max_fails=3 fail_timeout=30s;

    # max_fails: 3회 실패 시 제외
    # fail_timeout: 30초 후 재시도
}

# Active Health Check (nginx Plus 또는 모듈 필요)
upstream backend {
    zone backend 64k;

    server 192.168.1.101:3000;
    server 192.168.1.102:3000;
    server 192.168.1.103:3000;
}

server {
    location / {
        proxy_pass http://backend;
        health_check interval=5s fails=3 passes=2;
        # 5초마다 체크
        # 3회 실패 시 제외
        # 2회 성공 시 복귀
    }
}
```

## 31.4 AWS ALB (Application Load Balancer)

### Terraform 설정

```hcl
# ALB 생성
resource "aws_lb" "main" {
  name               = "app-lb"
  internal           = false
  load_balancer_type = "application"
  security_groups    = [aws_security_group.lb.id]
  subnets            = aws_subnet.public[*].id

  enable_deletion_protection = false
  enable_http2              = true

  tags = {
    Name = "app-lb"
  }
}

# Target Group
resource "aws_lb_target_group" "app" {
  name     = "app-tg"
  port     = 3000
  protocol = "HTTP"
  vpc_id   = aws_vpc.main.id

  # Health Check
  health_check {
    enabled             = true
    healthy_threshold   = 2
    unhealthy_threshold = 3
    timeout             = 5
    interval            = 30
    path                = "/health"
    matcher             = "200"
  }

  # Sticky Session
  stickiness {
    type            = "lb_cookie"
    cookie_duration = 86400  # 1일
    enabled         = true
  }

  # Deregistration Delay
  deregistration_delay = 30

  tags = {
    Name = "app-tg"
  }
}

# Target 등록 (EC2)
resource "aws_lb_target_group_attachment" "app" {
  count            = 3
  target_group_arn = aws_lb_target_group.app.arn
  target_id        = aws_instance.app[count.index].id
  port             = 3000
}

# Listener (HTTP → HTTPS 리다이렉트)
resource "aws_lb_listener" "http" {
  load_balancer_arn = aws_lb.main.arn
  port              = 80
  protocol          = "HTTP"

  default_action {
    type = "redirect"

    redirect {
      port        = "443"
      protocol    = "HTTPS"
      status_code = "HTTP_301"
    }
  }
}

# Listener (HTTPS)
resource "aws_lb_listener" "https" {
  load_balancer_arn = aws_lb.main.arn
  port              = 443
  protocol          = "HTTPS"
  ssl_policy        = "ELBSecurityPolicy-TLS-1-2-2017-01"
  certificate_arn   = aws_acm_certificate.main.arn

  default_action {
    type             = "forward"
    target_group_arn = aws_lb_target_group.app.arn
  }
}

# Listener Rule (경로 기반 라우팅)
resource "aws_lb_listener_rule" "api" {
  listener_arn = aws_lb_listener.https.arn
  priority     = 100

  condition {
    path_pattern {
      values = ["/api/*"]
    }
  }

  action {
    type             = "forward"
    target_group_arn = aws_lb_target_group.api.arn
  }
}
```

### 라우팅 전략

```hcl
# 1. Path-based Routing
resource "aws_lb_listener_rule" "static" {
  listener_arn = aws_lb_listener.https.arn
  priority     = 50

  condition {
    path_pattern {
      values = ["/static/*"]
    }
  }

  action {
    type             = "forward"
    target_group_arn = aws_lb_target_group.static.arn
  }
}

# 2. Host-based Routing
resource "aws_lb_listener_rule" "api_subdomain" {
  listener_arn = aws_lb_listener.https.arn
  priority     = 60

  condition {
    host_header {
      values = ["api.example.com"]
    }
  }

  action {
    type             = "forward"
    target_group_arn = aws_lb_target_group.api.arn
  }
}

# 3. Header-based Routing
resource "aws_lb_listener_rule" "mobile" {
  listener_arn = aws_lb_listener.https.arn
  priority     = 70

  condition {
    http_header {
      http_header_name = "User-Agent"
      values           = ["*Mobile*"]
    }
  }

  action {
    type             = "forward"
    target_group_arn = aws_lb_target_group.mobile.arn
  }
}

# 4. Weighted Target Groups (A/B 테스트)
resource "aws_lb_listener_rule" "ab_test" {
  listener_arn = aws_lb_listener.https.arn
  priority     = 80

  condition {
    path_pattern {
      values = ["/"]
    }
  }

  action {
    type = "forward"

    forward {
      target_group {
        arn    = aws_lb_target_group.version_a.arn
        weight = 90  # 90%
      }

      target_group {
        arn    = aws_lb_target_group.version_b.arn
        weight = 10  # 10%
      }

      stickiness {
        enabled  = true
        duration = 3600
      }
    }
  }
}
```

### Shared Session Storage

```javascript
// Redis 기반 세션 공유
const express = require('express');
const session = require('express-session');
const RedisStore = require('connect-redis').default;
const { createClient } = require('redis');

const app = express();

// Redis 클라이언트 (모든 서버가 같은 Redis 사용)
const redisClient = createClient({
  host: 'redis.example.com',
  port: 6379
});

redisClient.connect();

// 세션 미들웨어
app.use(session({
  store: new RedisStore({ client: redisClient }),
  secret: process.env.SESSION_SECRET,
  resave: false,
  saveUninitialized: false,
  cookie: {
    maxAge: 3600000,
    httpOnly: true,
    secure: true
  }
}));

// 어떤 서버든 세션 공유됨
app.get('/api/profile', (req, res) => {
  if (!req.session.userId) {
    return res.status(401).json({ error: '인증 필요' });
  }

  res.json({ userId: req.session.userId });
});

app.listen(3000);
```

## 31.6 Health Check

### Express Health Check

```javascript
const express = require('express');
const app = express();

// Health Check 엔드포인트
app.get('/health', (req, res) => {
  // 기본 헬스 체크
  res.status(200).json({ status: 'ok' });
});

// 상세 Health Check
app.get('/health/detailed', async (req, res) => {
  const checks = {
    server: 'ok',
    database: 'unknown',
    redis: 'unknown',
    memory: 'ok'
  };

  // DB 연결 체크
  try {
    await db.query('SELECT 1');
    checks.database = 'ok';
  } catch (err) {
    checks.database = 'error';
  }

  // Redis 체크
  try {
    await redis.ping();
    checks.redis = 'ok';
  } catch (err) {
    checks.redis = 'error';
  }

  // 메모리 체크
  const used = process.memoryUsage();
  if (used.heapUsed / used.heapTotal > 0.9) {
    checks.memory = 'warning';
  }

  // 하나라도 에러면 503
  const hasError = Object.values(checks).includes('error');
  const statusCode = hasError ? 503 : 200;

  res.status(statusCode).json(checks);
});

app.listen(3000);
```

### Nginx Health Check

```nginx
upstream backend {
```

## 31.8 Connection Draining

### Graceful Shutdown

```javascript
// Express Graceful Shutdown
const express = require('express');
const app = express();

const server = app.listen(3000);

// 종료 시그널 처리
process.on('SIGTERM', gracefulShutdown);
process.on('SIGINT', gracefulShutdown);

function gracefulShutdown() {
  console.log('Received shutdown signal');

  // 새 요청 거부
  server.close(() => {
    console.log('HTTP server closed');

    // DB 연결 종료
    db.close()
      .then(() => {
        console.log('Database connection closed');
        process.exit(0);
      })
      .catch(err => {
        console.error('Error during shutdown:', err);
        process.exit(1);
      });
  });

  // 30초 후 강제 종료
  setTimeout(() => {
    console.error('Forced shutdown');
    process.exit(1);
  }, 30000);
}
```

### AWS ALB Deregistration Delay

```hcl
resource "aws_lb_target_group" "app" {
  name = "app-tg"

  # Connection Draining
  deregistration_delay = 30  # 30초

  # 인스턴스 제거 시:
  # 1. 새 요청 안 받음
  # 2. 기존 연결 30초 대기
  # 3. 30초 후 강제 종료
}
```
