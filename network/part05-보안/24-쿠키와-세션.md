# Chapter 24. 쿠키와 세션

## 24.1 HTTP의 무상태성 문제

### Stateless의 문제

```
HTTP는 무상태 (Stateless):
각 요청이 독립적

문제:
요청 1: 로그인 (철수)
요청 2: 장바구니 조회
         → "당신이 누구죠?" (서버가 기억 못 함)

해결책: 쿠키와 세션
```

## 24.2 쿠키 (Cookie)

### 정의

```
쿠키: 브라우저에 저장되는 작은 텍스트 데이터

비유: 손등 도장
┌────────────────────────────┐
│ 놀이공원 입장               │
│ → 손등에 도장 찍기          │
│ → 재입장 시 도장 확인       │
└────────────────────────────┘
```

### 동작 원리

```
1. 서버가 쿠키 전송
   Set-Cookie: userId=123; sessionId=abc

2. 브라우저가 저장

3. 이후 요청마다 자동 전송
   Cookie: userId=123; sessionId=abc
```

### Express 쿠키 설정

```javascript
const express = require('express');
const cookieParser = require('cookie-parser');

const app = express();
app.use(cookieParser());

// 쿠키 설정
app.get('/set-cookie', (req, res) => {
  res.cookie('username', '철수', {
    maxAge: 24 * 60 * 60 * 1000,  // 1일
    httpOnly: true,                // JS 접근 차단
    secure: true,                  // HTTPS only
    sameSite: 'strict'             // CSRF 방지
  });

  res.send('쿠키 설정 완료');
});

// 쿠키 읽기
app.get('/get-cookie', (req, res) => {
  const username = req.cookies.username;
  res.send(`안녕하세요, ${username}님`);
});

// 쿠키 삭제
app.get('/clear-cookie', (req, res) => {
  res.clearCookie('username');
  res.send('쿠키 삭제 완료');
});
```

### 쿠키 옵션

```javascript
res.cookie('name', 'value', {
  // 유효 기간
  maxAge: 3600000,        // 밀리초 (1시간)
  expires: new Date('2025-12-31'),  // 만료일

  // 보안
  httpOnly: true,         // JS 접근 차단 (XSS 방어)
  secure: true,           // HTTPS만
  sameSite: 'strict',     // CSRF 방어
                          // strict | lax | none

  // 경로/도메인
  path: '/',              // 경로
  domain: '.example.com'  // 서브도메인 포함
});
```

## 24.3 세션 (Session)

### 정의

```
세션: 서버에 저장되는 사용자 정보

쿠키 vs 세션:
┌──────────┬──────────┬──────────┐
│   항목    │  쿠키    │  세션    │
├──────────┼──────────┼──────────┤
│ 저장 위치 │ 브라우저 │ 서버     │
│ 보안성    │ 낮음     │ 높음     │
│ 용량      │ 4KB      │ 제한없음 │
│ 속도      │ 빠름     │ 느림     │
└──────────┴──────────┴──────────┘
```

### Express 세션

```javascript
const session = require('express-session');
const RedisStore = require('connect-redis').default;
const { createClient } = require('redis');

// Redis 클라이언트
const redisClient = createClient({
  host: 'localhost',
  port: 6379
});
redisClient.connect();

// 세션 미들웨어
app.use(session({
  store: new RedisStore({ client: redisClient }),
  secret: 'your-secret-key',        // 서명 키
  resave: false,
  saveUninitialized: false,
  cookie: {
    maxAge: 30 * 60 * 1000,         // 30분
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'strict'
  }
}));

// 로그인: 세션 생성
app.post('/login', async (req, res) => {
  const { email, password } = req.body;

  const user = await User.findOne({ where: { email } });
  const isValid = await bcrypt.compare(password, user.password);

  if (!isValid) {
    return res.status(401).json({ error: '인증 실패' });
  }

  // 세션에 사용자 정보 저장
  req.session.userId = user.id;
  req.session.email = user.email;
  req.session.role = user.role;

  res.json({ message: '로그인 성공' });
});

// 세션 확인
app.get('/profile', (req, res) => {
  if (!req.session.userId) {
    return res.status(401).json({ error: '로그인 필요' });
  }

  res.json({
    userId: req.session.userId,
    email: req.session.email
  });
});

// 로그아웃: 세션 삭제
app.post('/logout', (req, res) => {
  req.session.destroy((err) => {
    if (err) {
      return res.status(500).json({ error: '로그아웃 실패' });
    }

    res.clearCookie('connect.sid');  // 세션 쿠키 삭제
    res.json({ message: '로그아웃 성공' });
  });
});
```

## 24.4 세션 저장소

### 메모리 (기본)

```javascript
// 개발 환경용 (재시작 시 소실)
app.use(session({
  secret: 'secret',
  resave: false,
  saveUninitialized: false
}));

⚠️ 프로덕션 사용 금지:
- 서버 재시작 시 모든 세션 소실
- 다중 서버 환경에서 동작 안 함
```

### Redis (권장)

```javascript
const RedisStore = require('connect-redis').default;
const { createClient } = require('redis');

const redisClient = createClient({
  socket: {
    host: process.env.REDIS_HOST || 'localhost',
    port: process.env.REDIS_PORT || 6379
  },
  password: process.env.REDIS_PASSWORD
});

redisClient.connect().catch(console.error);

app.use(session({
  store: new RedisStore({ client: redisClient }),
  secret: process.env.SESSION_SECRET,
  resave: false,
  saveUninitialized: false,
  cookie: {
    maxAge: 30 * 60 * 1000,
    httpOnly: true,
    secure: true,
    sameSite: 'strict'
  }
}));

장점:
✅ 빠름 (인메모리 DB)
✅ 영속성 (재시작해도 유지 가능)
✅ 다중 서버 환경 지원
```

### PostgreSQL/MongoDB

```javascript
// PostgreSQL
const pgSession = require('connect-pg-simple')(session);

app.use(session({
  store: new pgSession({
    conString: process.env.DATABASE_URL,
    tableName: 'sessions'
  }),
  secret: 'secret'
}));

// MongoDB
const MongoStore = require('connect-mongo');

app.use(session({
  store: MongoStore.create({
    mongoUrl: process.env.MONGO_URL,
    collectionName: 'sessions'
  }),
  secret: 'secret'
}));
```

## 24.5 세션 보안

### 세션 고정 공격 방지

```javascript
// 로그인 성공 시 세션 ID 재생성
app.post('/login', async (req, res) => {
  // 인증 로직...

  // 기존 세션 ID 폐기, 새 ID 생성
  req.session.regenerate((err) => {
    if (err) {
      return res.status(500).json({ error: '세션 생성 실패' });
    }

    req.session.userId = user.id;
    res.json({ message: '로그인 성공' });
  });
});
```

### 세션 하이재킹 방지

```javascript
// IP 주소 기록 및 검증
app.use((req, res, next) => {
  if (req.session.userId) {
    // 첫 로그인 시 IP 저장
    if (!req.session.ipAddress) {
      req.session.ipAddress = req.ip;
    }

    // IP 주소 변경 감지
    if (req.session.ipAddress !== req.ip) {
      req.session.destroy();
      return res.status(401).json({ error: '세션 무효화됨' });
    }
  }

  next();
});

// User-Agent 검증
app.use((req, res, next) => {
  if (req.session.userId) {
    if (!req.session.userAgent) {
      req.session.userAgent = req.headers['user-agent'];
    }

    if (req.session.userAgent !== req.headers['user-agent']) {
      req.session.destroy();
      return res.status(401).json({ error: '세션 무효화됨' });
    }
  }

  next();
});
```

### 세션 타임아웃

```javascript
// 비활동 시간 추적
app.use((req, res, next) => {
  if (req.session.userId) {
    const now = Date.now();
    const lastActivity = req.session.lastActivity || now;
    const timeout = 30 * 60 * 1000;  // 30분

    if (now - lastActivity > timeout) {
      req.session.destroy();
      return res.status(401).json({ error: '세션 만료' });
    }

    req.session.lastActivity = now;
  }

  next();
});
```

## 24.6 SameSite 쿠키

### SameSite 옵션

```javascript
// Strict: 가장 엄격
res.cookie('session', 'value', {
  sameSite: 'strict'
});
// 같은 사이트에서만 쿠키 전송
// 링크 클릭해서 들어올 때도 쿠키 안 보냄

// Lax: 중간 (기본값)
res.cookie('session', 'value', {
  sameSite: 'lax'
});
// 링크 클릭은 OK, POST는 NO

// None: 제한 없음
res.cookie('session', 'value', {
  sameSite: 'none',
  secure: true  // None은 Secure 필수
});
// 모든 경우 쿠키 전송
```

### CSRF 방어

```
SameSite=Strict/Lax로 CSRF 공격 방어

공격 시나리오:
1. 사용자가 bank.com 로그인 (쿠키 저장)
2. 악성 사이트 방문
3. 악성 사이트에서 bank.com으로 요청
   → SameSite 쿠키는 전송 안 됨 ✅
```

## 24.7 실전 인증 미들웨어

```javascript
// 인증 필수
function authenticate(req, res, next) {
  if (!req.session.userId) {
    return res.status(401).json({
      error: '인증 필요',
      code: 'UNAUTHORIZED'
    });
  }

  next();
}

// 선택적 인증 (있으면 사용)
function optionalAuth(req, res, next) {
  // 세션 있으면 사용자 정보 로드
  if (req.session.userId) {
    User.findByPk(req.session.userId)
      .then(user => {
        req.user = user;
        next();
      });
  } else {
    next();
  }
}

// 사용
app.get('/api/profile', authenticate, async (req, res) => {
  const user = await User.findByPk(req.session.userId);
  res.json(user);
});

app.get('/api/posts', optionalAuth, async (req, res) => {
  // req.user 있으면 개인화된 콘텐츠
  // 없으면 일반 콘텐츠
});
```

## 24.8 Remember Me 기능

```javascript
const crypto = require('crypto');

// 로그인 시
app.post('/login', async (req, res) => {
  const { email, password, rememberMe } = req.body;

  // 인증...

  req.session.userId = user.id;

  if (rememberMe) {
    // Remember Token 생성
    const token = crypto.randomBytes(32).toString('hex');

    await RememberToken.create({
      userId: user.id,
      token,
      expiresAt: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000)  // 30일
    });

    // 쿠키로 전송
    res.cookie('remember_token', token, {
      maxAge: 30 * 24 * 60 * 60 * 1000,
      httpOnly: true,
      secure: true,
      sameSite: 'strict'
    });
  }

  res.json({ message: '로그인 성공' });
});

// Remember Token 검증 미들웨어
app.use(async (req, res, next) => {
  // 이미 세션 있으면 패스
  if (req.session.userId) {
    return next();
  }

  // Remember Token 확인
  const token = req.cookies.remember_token;
  if (!token) {
    return next();
  }

  const rememberToken = await RememberToken.findOne({
    where: { token },
    include: User
  });

  if (!rememberToken || rememberToken.expiresAt < new Date()) {
    res.clearCookie('remember_token');
    return next();
  }

  // 자동 로그인
  req.session.userId = rememberToken.userId;
  next();
});
```
