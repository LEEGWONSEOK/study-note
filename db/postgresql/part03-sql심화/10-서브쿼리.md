# Chapter 10. 서브쿼리

## 10.1 서브쿼리란?

### 정의

```
서브쿼리 (Subquery):
다른 쿼리 안에 포함된 쿼리

비유: 문제 풀이의 중간 계산
┌────────────────────────────────┐
│ 1. 평균 나이 계산 (서브쿼리)   │
│    → 30세                      │
│                                │
│ 2. 평균보다 나이 많은 사람     │
│    (메인 쿼리)                 │
│    → WHERE age > 30            │
└────────────────────────────────┘
```

### 서브쿼리 위치

```sql
-- SELECT 절
SELECT
  name,
  (SELECT COUNT(*) FROM posts WHERE user_id = users.id) AS post_count
FROM users;

-- FROM 절
SELECT *
FROM (SELECT * FROM users WHERE age > 20) AS young_users;

-- WHERE 절
SELECT * FROM users
WHERE age > (SELECT AVG(age) FROM users);
```

## 10.2 WHERE 절 서브쿼리

### 단일 값 반환

```sql
-- 평균 나이보다 많은 사용자
SELECT name, age
FROM users
WHERE age > (SELECT AVG(age) FROM users);

-- 최근 가입자보다 먼저 가입
SELECT name, created_at
FROM users
WHERE created_at < (
  SELECT MAX(created_at) FROM users WHERE country = 'Korea'
);
```

### IN 서브쿼리

```sql
-- 글을 쓴 사용자만
SELECT name
FROM users
WHERE id IN (SELECT DISTINCT user_id FROM posts);

-- 특정 카테고리 글 작성자
SELECT name
FROM users
WHERE id IN (
  SELECT user_id
  FROM posts
  WHERE category_id IN (1, 2, 3)
);
```

### NOT IN

```sql
-- 글을 한 번도 안 쓴 사용자
SELECT name
FROM users
WHERE id NOT IN (SELECT user_id FROM posts WHERE user_id IS NOT NULL);
-- ⚠️ NULL 주의! user_id가 NULL 있으면 결과 없음

-- 안전한 방법
SELECT name
FROM users
WHERE id NOT IN (
  SELECT user_id FROM posts WHERE user_id IS NOT NULL
);

-- 또는 NOT EXISTS 사용 (권장)
SELECT name
FROM users u
WHERE NOT EXISTS (
  SELECT 1 FROM posts p WHERE p.user_id = u.id
);
```

### EXISTS

```sql
-- 글이 있는 사용자
SELECT name
FROM users u
WHERE EXISTS (
  SELECT 1 FROM posts p WHERE p.user_id = u.id
);

-- 최근 7일 내 활동한 사용자
SELECT name
FROM users u
WHERE EXISTS (
  SELECT 1
  FROM posts p
  WHERE p.user_id = u.id
    AND p.created_at >= NOW() - INTERVAL '7 days'
);
```

### ANY / ALL

```sql
-- ANY: 하나라도 만족
SELECT name, age
FROM users
WHERE age > ANY (SELECT age FROM users WHERE country = 'Korea');
-- 한국 사용자 중 가장 어린 사람보다 나이 많으면

-- ALL: 모두 만족
SELECT name, age
FROM users
WHERE age > ALL (SELECT age FROM users WHERE country = 'Korea');
-- 한국 사용자 모두보다 나이 많으면

-- 동일한 의미
WHERE age > ANY (...) = WHERE age > (SELECT MIN(age) FROM ...)
WHERE age > ALL (...) = WHERE age > (SELECT MAX(age) FROM ...)
```

## 10.3 SELECT 절 서브쿼리

### 스칼라 서브쿼리

```sql
-- 사용자별 게시글 수
SELECT
  name,
  email,
  (SELECT COUNT(*) FROM posts WHERE user_id = users.id) AS post_count
FROM users;

-- 결과:
--   name   |      email       | post_count
-- ---------+------------------+------------
--  Alice   | alice@...        | 5
--  Bob     | bob@...          | 2
--  Charlie | charlie@...      | 0
```

### 여러 서브쿼리

```sql
SELECT
  u.name,
  (SELECT COUNT(*) FROM posts WHERE user_id = u.id) AS post_count,
  (SELECT COUNT(*) FROM comments WHERE user_id = u.id) AS comment_count,
  (SELECT MAX(created_at) FROM posts WHERE user_id = u.id) AS last_post_date
FROM users u;
```

### 조건부 서브쿼리

```sql
SELECT
  p.title,
  p.view_count,
  (
    SELECT AVG(view_count)
    FROM posts
    WHERE category_id = p.category_id
  ) AS category_avg_views
FROM posts p;
```

## 10.4 FROM 절 서브쿼리 (인라인 뷰)

### 기본 사용

```sql
-- 서브쿼리를 테이블처럼 사용
SELECT *
FROM (
  SELECT name, age
  FROM users
  WHERE age > 20
) AS young_users
WHERE age < 30;

-- 20대만
```

### 집계 후 필터링

```sql
-- 게시글 3개 이상인 사용자
SELECT user_id, post_count
FROM (
  SELECT
    user_id,
    COUNT(*) AS post_count
  FROM posts
  GROUP BY user_id
) AS user_posts
WHERE post_count >= 3;
```

### JOIN과 조합

```sql
-- 인기 게시글 작성자
SELECT
  u.name,
  popular_posts.post_count
FROM users u
INNER JOIN (
  SELECT
    user_id,
    COUNT(*) AS post_count
  FROM posts
  WHERE view_count > 1000
  GROUP BY user_id
) AS popular_posts ON u.id = popular_posts.user_id;
```

## 10.5 상관 서브쿼리

### 정의

```
상관 서브쿼리 (Correlated Subquery):
외부 쿼리의 컬럼을 참조하는 서브쿼리
```

### 예제

```sql
-- 카테고리 평균보다 조회수 많은 게시글
SELECT
  p.title,
  p.view_count,
  p.category_id
FROM posts p
WHERE p.view_count > (
  SELECT AVG(view_count)
  FROM posts
  WHERE category_id = p.category_id  -- 외부 쿼리 참조!
);
```

### 성능 주의

```sql
-- ❌ 느림: 상관 서브쿼리
SELECT
  name,
  (SELECT COUNT(*) FROM posts WHERE user_id = users.id) AS post_count
FROM users;
-- 사용자마다 서브쿼리 실행

-- ✅ 빠름: JOIN
SELECT
  u.name,
  COUNT(p.id) AS post_count
FROM users u
LEFT JOIN posts p ON u.id = p.user_id
GROUP BY u.id, u.name;
```

## 10.6 WITH (CTE)

### 기본 문법

```sql
WITH cte_name AS (
  SELECT ...
)
SELECT * FROM cte_name;
```

### 단일 CTE

```sql
-- 평균 나이 계산 후 사용
WITH avg_age_cte AS (
  SELECT AVG(age) AS avg_age
  FROM users
)
SELECT name, age
FROM users, avg_age_cte
WHERE age > avg_age_cte.avg_age;
```

### 여러 CTE

```sql
WITH
  user_stats AS (
    SELECT
      user_id,
      COUNT(*) AS post_count
    FROM posts
    GROUP BY user_id
  ),
  active_users AS (
    SELECT user_id
    FROM user_stats
    WHERE post_count >= 5
  )
SELECT
  u.name,
  us.post_count
FROM users u
INNER JOIN user_stats us ON u.id = us.user_id
WHERE u.id IN (SELECT user_id FROM active_users);
```

### 재귀 CTE

```sql
-- 1부터 10까지
WITH RECURSIVE numbers AS (
  -- 기본 케이스
  SELECT 1 AS n
  UNION ALL
  -- 재귀 케이스
  SELECT n + 1
  FROM numbers
  WHERE n < 10
)
SELECT * FROM numbers;

-- 결과: 1, 2, 3, ..., 10
```

### 조직도 (재귀)

```sql
-- 특정 매니저 하위 모든 직원
WITH RECURSIVE employee_tree AS (
  -- 시작점: CTO
  SELECT id, name, manager_id, 1 AS level
  FROM employees
  WHERE id = 2  -- CTO

  UNION ALL

  -- 재귀: 하위 직원
  SELECT e.id, e.name, e.manager_id, et.level + 1
  FROM employees e
  INNER JOIN employee_tree et ON e.manager_id = et.id
)
SELECT * FROM employee_tree
ORDER BY level, id;
```

## 10.7 실전 예제

### TOP N per Group

```sql
-- 카테고리별 인기 게시글 TOP 3
WITH ranked_posts AS (
  SELECT
    *,
    ROW_NUMBER() OVER (PARTITION BY category_id ORDER BY view_count DESC) AS rn
  FROM posts
)
SELECT
  category_id,
  title,
  view_count
FROM ranked_posts
WHERE rn <= 3
ORDER BY category_id, rn;
```

### 이동 평균

```sql
-- 7일 이동 평균 가입자 수
WITH daily_signups AS (
  SELECT
    DATE(created_at) AS signup_date,
    COUNT(*) AS signups
  FROM users
  GROUP BY DATE(created_at)
)
SELECT
  signup_date,
  signups,
  AVG(signups) OVER (
    ORDER BY signup_date
    ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
  ) AS moving_avg_7d
FROM daily_signups
ORDER BY signup_date DESC;
```

### 중복 데이터 찾기

```sql
-- 이메일 중복
SELECT email, COUNT(*) AS count
FROM users
GROUP BY email
HAVING COUNT(*) > 1;

-- 중복 제거 (최신만 유지)
WITH duplicates AS (
  SELECT
    id,
    ROW_NUMBER() OVER (PARTITION BY email ORDER BY created_at DESC) AS rn
  FROM users
)
DELETE FROM users
WHERE id IN (
  SELECT id FROM duplicates WHERE rn > 1
);
```

## 10.8 서브쿼리 최적화

### 서브쿼리 → JOIN

```sql
-- ❌ 느림: 서브쿼리
SELECT *
FROM users
WHERE id IN (SELECT user_id FROM posts WHERE created_at > '2024-01-01');

-- ✅ 빠름: JOIN
SELECT DISTINCT u.*
FROM users u
INNER JOIN posts p ON u.id = p.user_id
WHERE p.created_at > '2024-01-01';
```

### EXISTS vs IN

```sql
-- EXISTS (일반적으로 더 빠름)
SELECT name
FROM users u
WHERE EXISTS (
  SELECT 1 FROM posts p WHERE p.user_id = u.id
);

-- IN
SELECT name
FROM users
WHERE id IN (SELECT user_id FROM posts);

-- EXISTS는 첫 매칭에서 중단
-- IN은 전체 목록 생성
```

### NOT EXISTS vs NOT IN

```sql
-- ✅ NOT EXISTS (NULL 안전, 빠름)
SELECT name
FROM users u
WHERE NOT EXISTS (
  SELECT 1 FROM posts p WHERE p.user_id = u.id
);

-- ❌ NOT IN (NULL 문제, 느림)
SELECT name
FROM users
WHERE id NOT IN (SELECT user_id FROM posts);
-- user_id에 NULL 있으면 결과 없음!
```

## 10.9 윈도우 함수 vs 서브쿼리

### 서브쿼리 방식

```sql
-- 사용자 + 카테고리별 평균 조회수
SELECT
  p.title,
  p.view_count,
  (
    SELECT AVG(view_count)
    FROM posts
    WHERE category_id = p.category_id
  ) AS category_avg
FROM posts p;
```

### 윈도우 함수 (더 빠름)

```sql
-- 동일한 결과, 더 빠름
SELECT
  title,
  view_count,
  AVG(view_count) OVER (PARTITION BY category_id) AS category_avg
FROM posts;
```

## 10.10 복잡한 예제

### 연속 방문자

```sql
-- 3일 연속 로그인한 사용자
WITH login_dates AS (
  SELECT DISTINCT
    user_id,
    DATE(login_at) AS login_date
  FROM login_logs
),
streaks AS (
  SELECT
    user_id,
    login_date,
    login_date - ROW_NUMBER() OVER (
      PARTITION BY user_id
      ORDER BY login_date
    )::INT AS streak_group
  FROM login_dates
)
SELECT
  user_id,
  COUNT(*) AS streak_length,
  MIN(login_date) AS streak_start,
  MAX(login_date) AS streak_end
FROM streaks
GROUP BY user_id, streak_group
HAVING COUNT(*) >= 3;
```

### 매출 성장률

```sql
-- 월별 매출 + 전월 대비 성장률
WITH monthly_revenue AS (
  SELECT
    DATE_TRUNC('month', order_date) AS month,
    SUM(total_amount) AS revenue
  FROM orders
  WHERE status = 'completed'
  GROUP BY DATE_TRUNC('month', order_date)
)
SELECT
  month,
  revenue,
  LAG(revenue) OVER (ORDER BY month) AS prev_month_revenue,
  ROUND(
    (revenue - LAG(revenue) OVER (ORDER BY month))
    / LAG(revenue) OVER (ORDER BY month) * 100,
    2
  ) AS growth_rate
FROM monthly_revenue
ORDER BY month DESC;
```

## 핵심 요약

### 서브쿼리 위치
```sql
SELECT (서브쿼리) FROM table;           -- SELECT 절
SELECT * FROM (서브쿼리) AS alias;      -- FROM 절
SELECT * FROM table WHERE col IN (서브쿼리);  -- WHERE 절
```

### 비교
```sql
IN / NOT IN          -- 목록 포함
EXISTS / NOT EXISTS  -- 존재 여부 (더 빠름)
ANY / ALL            -- 하나라도 / 모두
```

### CTE
```sql
WITH cte_name AS (
  SELECT ...
)
SELECT * FROM cte_name;
```

### 최적화
```sql
서브쿼리 → JOIN       -- 일반적으로 더 빠름
NOT IN → NOT EXISTS   -- NULL 안전, 빠름
상관 서브쿼리 최소화  -- 성능 저하
```
