# Chapter 18. 락과 동시성제어

## 18.1 락(Lock)이란?

### 정의

```
락 (Lock):
여러 트랜잭션이 동시에 같은 데이터를 수정하지 못하도록 제어하는 메커니즘

비유: 화장실
┌─────────────────────────────────┐
│ 1. 사용 중 (🔒 잠금)            │
│    - 다른 사람 대기             │
│    - 배타적 락 (Exclusive)      │
│                                 │
│ 2. 표지판만 읽기 (👀)           │
│    - 여러 명 동시 읽기 가능     │
│    - 공유 락 (Shared)           │
└─────────────────────────────────┘

락 = 데이터 보호 장치
```

### 락의 필요성

```sql
-- ❌ 락 없이 동시 업데이트
-- 세션 1
UPDATE accounts SET balance = balance - 100 WHERE id = 1;

-- 세션 2 (동시 실행)
UPDATE accounts SET balance = balance - 50 WHERE id = 1;

-- 결과: 잔액이 -50만 차감될 수 있음! (Lost Update)


-- ✅ 락으로 보호
-- 세션 1
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR UPDATE;
-- 락 획득
UPDATE accounts SET balance = balance - 100 WHERE id = 1;
COMMIT;

-- 세션 2
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR UPDATE;
-- 대기... (세션 1이 끝날 때까지)
UPDATE accounts SET balance = balance - 50 WHERE id = 1;
COMMIT;
-- 순차 실행으로 정확한 결과!
```

## 18.2 락의 종류

### 테이블 락 (Table Lock)

```sql
-- ACCESS SHARE (SELECT)
BEGIN;
SELECT * FROM users;
-- 다른 트랜잭션: SELECT 가능, 대부분 DML 가능
COMMIT;


-- ROW SHARE (SELECT FOR SHARE)
BEGIN;
SELECT * FROM users FOR SHARE;
-- 다른 트랜잭션: SELECT 가능, UPDATE/DELETE 대기
COMMIT;


-- ROW EXCLUSIVE (INSERT, UPDATE, DELETE)
BEGIN;
UPDATE users SET name = 'Alice' WHERE id = 1;
-- 다른 트랜잭션: SELECT 가능, 쓰기 대기
COMMIT;


-- SHARE (CREATE INDEX)
BEGIN;
LOCK TABLE users IN SHARE MODE;
-- 다른 트랜잭션: SELECT 가능, 쓰기 불가
COMMIT;


-- EXCLUSIVE (REFRESH MATERIALIZED VIEW)
BEGIN;
LOCK TABLE users IN EXCLUSIVE MODE;
-- 다른 트랜잭션: SELECT만 가능
COMMIT;


-- ACCESS EXCLUSIVE (ALTER TABLE, DROP TABLE, VACUUM FULL)
BEGIN;
LOCK TABLE users IN ACCESS EXCLUSIVE MODE;
-- 다른 트랜잭션: 모든 작업 대기
COMMIT;
```

### 행 락 (Row Lock)

```sql
-- FOR UPDATE (배타적 락)
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR UPDATE;
-- 이 행에 대해 다른 트랜잭션의 UPDATE/DELETE 차단

UPDATE accounts SET balance = balance - 100 WHERE id = 1;
COMMIT;


-- FOR NO KEY UPDATE (외래키 컬럼 제외 배타적 락)
BEGIN;
SELECT * FROM users WHERE id = 1 FOR NO KEY UPDATE;
-- id 외 컬럼만 수정 가능 (외래키 참조는 허용)
COMMIT;


-- FOR SHARE (공유 락)
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR SHARE;
-- 다른 트랜잭션: 읽기 가능, 쓰기 대기
COMMIT;


-- FOR KEY SHARE (키 컬럼만 공유 락)
BEGIN;
SELECT * FROM users WHERE id = 1 FOR KEY SHARE;
-- 다른 트랜잭션: id는 보호, 다른 컬럼은 수정 가능
COMMIT;
```

## 18.3 FOR UPDATE

### 기본 사용

```sql
-- 읽는 순간 락 획득
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR UPDATE;
--  id | user_id | balance
-- ----+---------+---------
--   1 |    1    | 10000
-- 락 획득! 다른 트랜잭션은 대기

UPDATE accounts SET balance = balance - 1000 WHERE id = 1;
COMMIT;
```

### 여러 행 잠금

```sql
BEGIN;
SELECT * FROM accounts WHERE user_id = 1 FOR UPDATE;
-- user_id=1인 모든 계좌 잠금

UPDATE accounts SET balance = balance * 1.05 WHERE user_id = 1;
COMMIT;
```

### 조인 시 FOR UPDATE

```sql
-- 특정 테이블만 잠금
BEGIN;
SELECT u.name, a.balance
FROM users u
JOIN accounts a ON u.id = a.user_id
WHERE u.id = 1
FOR UPDATE OF a;  -- accounts 테이블만 잠금

UPDATE accounts SET balance = balance - 1000 WHERE user_id = 1;
COMMIT;
```

## 18.4 FOR UPDATE 옵션

### NOWAIT (대기 안 함)

```sql
-- 세션 1
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR UPDATE;
-- 락 획득

-- 세션 2
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR UPDATE NOWAIT;
-- ERROR: could not obtain lock on row in relation "accounts"
-- 즉시 에러 반환!

ROLLBACK;

-- 애플리케이션에서 재시도 로직
DO $$
DECLARE
  account_row RECORD;
  retry_count INT := 0;
BEGIN
  LOOP
    BEGIN
      SELECT * INTO account_row
      FROM accounts WHERE id = 1
      FOR UPDATE NOWAIT;

      -- 성공
      EXIT;
    EXCEPTION
      WHEN lock_not_available THEN
        retry_count := retry_count + 1;
        IF retry_count >= 5 THEN
          RAISE EXCEPTION 'Failed to acquire lock';
        END IF;
        PERFORM pg_sleep(0.1);
    END;
  END LOOP;

  -- 작업 수행
  UPDATE accounts SET balance = balance - 1000 WHERE id = 1;
END $$;
```

### SKIP LOCKED (잠긴 행 건너뛰기)

```sql
-- 작업 큐 구현
CREATE TABLE tasks (
  id SERIAL PRIMARY KEY,
  status VARCHAR(20) DEFAULT 'pending',
  worker_id INT,
  data TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Worker 1
BEGIN;
UPDATE tasks
SET status = 'processing', worker_id = 1
WHERE id = (
  SELECT id FROM tasks
  WHERE status = 'pending'
  ORDER BY created_at
  LIMIT 1
  FOR UPDATE SKIP LOCKED  -- 잠긴 작업은 건너뛰기
)
RETURNING *;
--  id | status     | worker_id | data
-- ----+------------+-----------+------
--  10 | processing | 1         | ...

-- 작업 처리...
UPDATE tasks SET status = 'completed' WHERE id = 10;
COMMIT;


-- Worker 2 (동시 실행)
BEGIN;
UPDATE tasks
SET status = 'processing', worker_id = 2
WHERE id = (
  SELECT id FROM tasks
  WHERE status = 'pending'
  ORDER BY created_at
  LIMIT 1
  FOR UPDATE SKIP LOCKED
)
RETURNING *;
--  id | status     | worker_id | data
-- ----+------------+-----------+------
--  11 | processing | 2         | ...
-- 다른 작업을 가져감!

COMMIT;
```

### SKIP LOCKED 활용 패턴

```sql
-- 패턴 1: 분산 작업 처리
-- 여러 워커가 동시에 작업 가져가기
SELECT * FROM jobs
WHERE status = 'pending'
ORDER BY priority DESC, created_at
LIMIT 10
FOR UPDATE SKIP LOCKED;


-- 패턴 2: 티켓팅 (선착순 좌석)
BEGIN;
UPDATE seats
SET is_booked = true, user_id = 1
WHERE id = (
  SELECT id FROM seats
  WHERE screening_id = 100 AND is_booked = false
  ORDER BY seat_number
  LIMIT 1
  FOR UPDATE SKIP LOCKED
)
RETURNING seat_number;
-- 'A1'
COMMIT;


-- 패턴 3: 재고 할당
BEGIN;
UPDATE inventory
SET allocated_qty = allocated_qty + 10, warehouse_id = 5
WHERE id = (
  SELECT id FROM inventory
  WHERE product_id = 100 AND available_qty >= 10
  ORDER BY warehouse_id
  LIMIT 1
  FOR UPDATE SKIP LOCKED
)
RETURNING *;
COMMIT;
```

## 18.5 FOR SHARE

### 공유 락

```sql
-- 읽기는 허용, 쓰기는 차단
-- 세션 1
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR SHARE;
-- 공유 락 획득

-- 세션 2
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR SHARE;
-- 가능! (공유 락은 여러 개 허용)

-- 세션 3
BEGIN;
UPDATE accounts SET balance = 5000 WHERE id = 1;
-- 대기... (공유 락이 모두 해제될 때까지)
```

### 실전 예제: 재고 확인 후 주문

```sql
-- 재고 확인 (공유 락)
BEGIN;
SELECT stock FROM products WHERE id = 10 FOR SHARE;
-- stock: 100

-- 다른 트랜잭션도 재고 확인 가능
-- 하지만 재고 변경은 대기

-- 주문 가능 여부 판단
IF stock >= 5 THEN
  -- 주문 생성
  INSERT INTO orders (user_id, product_id, quantity)
  VALUES (1, 10, 5);

  -- 재고 차감 (배타적 락으로 자동 업그레이드)
  UPDATE products SET stock = stock - 5 WHERE id = 10;
END IF;

COMMIT;
```

## 18.6 Advisory Lock (권고 락)

### 정의

```
Advisory Lock:
애플리케이션 레벨에서 정의하는 논리적 락
테이블/행이 아닌 숫자(ID)에 대한 락

특징:
- 트랜잭션 레벨 또는 세션 레벨
- 자동 해제 안 됨 (명시적 해제 필요)
- 유연한 잠금 제어
```

### 기본 사용

```sql
-- 트랜잭션 레벨 락
BEGIN;
SELECT pg_advisory_lock(123);
-- 123번 락 획득

-- 작업 수행...

COMMIT;
-- 자동 해제


-- 세션 레벨 락
SELECT pg_advisory_lock(456);
-- 456번 락 획득

-- 작업 수행...

-- 명시적 해제
SELECT pg_advisory_unlock(456);


-- NOWAIT 버전
SELECT pg_try_advisory_lock(789);
-- true (성공) / false (실패)
```

### 실전 예제: 중복 처리 방지

```sql
-- 배치 작업 중복 실행 방지
DO $$
BEGIN
  -- 락 획득 시도
  IF pg_try_advisory_lock(1001) THEN
    -- 배치 작업 실행
    RAISE NOTICE 'Starting batch job...';

    -- 실제 작업
    UPDATE users SET last_processed = NOW()
    WHERE status = 'pending';

    -- 락 해제
    PERFORM pg_advisory_unlock(1001);
  ELSE
    -- 이미 다른 프로세스가 실행 중
    RAISE NOTICE 'Batch job already running';
  END IF;
END $$;
```

### 실전 예제: 순차 번호 생성

```sql
-- 주문 번호 생성 (중복 방지)
CREATE OR REPLACE FUNCTION generate_order_number()
RETURNS TEXT AS $$
DECLARE
  lock_key CONSTANT BIGINT := 2001;
  today TEXT;
  seq_num INT;
  order_number TEXT;
BEGIN
  -- 락 획득
  PERFORM pg_advisory_lock(lock_key);

  -- 오늘 날짜
  today := TO_CHAR(NOW(), 'YYYYMMDD');

  -- 오늘의 마지막 순번 조회
  SELECT COALESCE(MAX(SUBSTRING(order_no FROM 10)::INT), 0) + 1
  INTO seq_num
  FROM orders
  WHERE order_no LIKE today || '%';

  -- 주문 번호 생성
  order_number := today || LPAD(seq_num::TEXT, 4, '0');

  -- 락 해제
  PERFORM pg_advisory_unlock(lock_key);

  RETURN order_number;
END;
$$ LANGUAGE plpgsql;

-- 사용
SELECT generate_order_number();
-- '202401150001'
```

## 18.7 데드락 (Deadlock)

### 발생 원인

```sql
-- 세션 1
BEGIN;
UPDATE accounts SET balance = balance - 100 WHERE id = 1;
-- LOCK on id=1

-- 세션 2
BEGIN;
UPDATE accounts SET balance = balance - 100 WHERE id = 2;
-- LOCK on id=2

-- 세션 1
UPDATE accounts SET balance = balance + 100 WHERE id = 2;
-- 대기... (세션 2가 id=2 잠금 중)

-- 세션 2
UPDATE accounts SET balance = balance + 100 WHERE id = 1;
-- 대기... (세션 1이 id=1 잠금 중)

-- 1초 후 PostgreSQL이 자동 감지
-- ERROR: deadlock detected
-- DETAIL: Process 1234 waits for ShareLock on transaction 5678
```

### 데드락 방지 전략

#### 1. 잠금 순서 일관성

```sql
-- ❌ 나쁜 예: 순서가 다름
-- 트랜잭션 1: LOCK(1) → LOCK(2)
-- 트랜잭션 2: LOCK(2) → LOCK(1)

-- ✅ 좋은 예: 순서 일관성
CREATE OR REPLACE FUNCTION transfer_money(
  from_id INT,
  to_id INT,
  amount NUMERIC
) RETURNS VOID AS $$
DECLARE
  min_id INT := LEAST(from_id, to_id);
  max_id INT := GREATEST(from_id, to_id);
BEGIN
  -- 항상 ID 순서대로 잠금
  PERFORM * FROM accounts WHERE id = min_id FOR UPDATE;
  PERFORM * FROM accounts WHERE id = max_id FOR UPDATE;

  -- 이체
  UPDATE accounts SET balance = balance - amount WHERE id = from_id;
  UPDATE accounts SET balance = balance + amount WHERE id = to_id;
END;
$$ LANGUAGE plpgsql;

-- 사용
BEGIN;
SELECT transfer_money(1, 2, 1000);
COMMIT;

BEGIN;
SELECT transfer_money(2, 1, 500);
COMMIT;
-- 데드락 발생 안 함!
```

#### 2. 잠금 시간 최소화

```sql
-- ❌ 나쁜 예: 트랜잭션이 너무 김
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR UPDATE;

-- 외부 API 호출 (느림!)
PERFORM http_post('https://api.example.com/notify');

UPDATE accounts SET balance = balance - 1000 WHERE id = 1;
COMMIT;


-- ✅ 좋은 예: 빠른 작업만 트랜잭션 안에
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR UPDATE;
UPDATE accounts SET balance = balance - 1000 WHERE id = 1;
COMMIT;

-- 외부 API는 트랜잭션 밖에서
PERFORM http_post('https://api.example.com/notify');
```

#### 3. NOWAIT 사용

```sql
-- 데드락 대신 즉시 실패
BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR UPDATE;

-- NOWAIT로 즉시 실패
BEGIN
  SELECT * FROM accounts WHERE id = 2 FOR UPDATE NOWAIT;
EXCEPTION
  WHEN lock_not_available THEN
    ROLLBACK;
    RAISE NOTICE 'Could not acquire lock, retrying...';
    -- 재시도 로직
END;
```

### 데드락 모니터링

```sql
-- 데드락 통계
SELECT
  datname,
  deadlocks
FROM pg_stat_database
WHERE datname = 'mydb';

--  datname | deadlocks
-- ---------+-----------
--  mydb    | 5

-- 로그 확인 (postgresql.conf)
-- log_lock_waits = on
-- deadlock_timeout = 1s

-- /var/log/postgresql/postgresql.log
-- ERROR: deadlock detected
-- DETAIL: Process 1234 waits for ShareLock on transaction 5678
```

## 18.8 락 타임아웃

### statement_timeout

```sql
-- 쿼리 실행 시간 제한 (밀리초)
SET statement_timeout = 5000;  -- 5초

BEGIN;
SELECT * FROM users WHERE id = 1 FOR UPDATE;
-- 5초 내에 락 획득 못하면
-- ERROR: canceling statement due to statement timeout

ROLLBACK;
```

### lock_timeout

```sql
-- 락 대기 시간 제한 (밀리초)
SET lock_timeout = 2000;  -- 2초

BEGIN;
SELECT * FROM accounts WHERE id = 1 FOR UPDATE;
-- 2초 내에 락 획득 못하면
-- ERROR: canceling statement due to lock timeout

ROLLBACK;
```

### idle_in_transaction_session_timeout

```sql
-- 트랜잭션 유휴 시간 제한 (밀리초)
SET idle_in_transaction_session_timeout = 60000;  -- 60초

BEGIN;
SELECT * FROM users WHERE id = 1;
-- 60초간 아무것도 안 하면
-- FATAL: terminating connection due to idle-in-transaction timeout

ROLLBACK;
```

## 18.9 락 모니터링

### 현재 락 확인

```sql
-- 활성 락 목록
SELECT
  l.locktype,
  l.relation::regclass AS table_name,
  l.mode,
  l.granted,
  a.pid,
  a.usename,
  a.query
FROM pg_locks l
LEFT JOIN pg_stat_activity a ON l.pid = a.pid
WHERE l.granted = true
ORDER BY l.pid;

--  locktype | table_name | mode             | granted | pid  | query
-- ----------+------------+------------------+---------+------+--------
--  relation | accounts   | RowExclusiveLock | t       | 1234 | UPDATE...
--  relation | users      | AccessShareLock  | t       | 5678 | SELECT...
```

### 대기 중인 락

```sql
-- 락 대기 상황
SELECT
  waiting.pid AS waiting_pid,
  waiting.query AS waiting_query,
  blocking.pid AS blocking_pid,
  blocking.query AS blocking_query,
  NOW() - waiting.query_start AS wait_duration
FROM pg_stat_activity AS waiting
JOIN pg_locks AS waiting_lock ON waiting.pid = waiting_lock.pid
JOIN pg_locks AS blocking_lock ON waiting_lock.relation = blocking_lock.relation
  AND waiting_lock.locktype = blocking_lock.locktype
JOIN pg_stat_activity AS blocking ON blocking_lock.pid = blocking.pid
WHERE NOT waiting_lock.granted
  AND blocking_lock.granted
  AND waiting.pid != blocking.pid;

--  waiting_pid | waiting_query    | blocking_pid | blocking_query | wait_duration
-- -------------+------------------+--------------+----------------+---------------
--  5678        | UPDATE accounts..| 1234         | UPDATE accounts| 00:00:05
```

### 락 차단 체인

```sql
-- 복잡한 락 대기 체인 조회
WITH RECURSIVE lock_tree AS (
  -- 루트: 차단하는 트랜잭션
  SELECT
    pid,
    pg_blocking_pids(pid) AS blocking_pids,
    query,
    1 AS level
  FROM pg_stat_activity
  WHERE cardinality(pg_blocking_pids(pid)) = 0
    AND state = 'active'

  UNION ALL

  -- 재귀: 차단당하는 트랜잭션
  SELECT
    a.pid,
    pg_blocking_pids(a.pid),
    a.query,
    lt.level + 1
  FROM pg_stat_activity a
  JOIN lock_tree lt ON a.pid = ANY(lt.blocking_pids)
)
SELECT
  level,
  pid,
  blocking_pids,
  LEFT(query, 50) AS query
FROM lock_tree
ORDER BY level, pid;
```

### 락 킬하기

```sql
-- 특정 프로세스 종료
SELECT pg_terminate_backend(1234);

-- 5분 이상 대기 중인 프로세스 종료
SELECT pg_terminate_backend(pid)
FROM pg_stat_activity
WHERE state = 'active'
  AND wait_event_type = 'Lock'
  AND NOW() - query_start > INTERVAL '5 minutes';
```

## 18.10 동시성 제어 패턴

### 패턴 1: 낙관적 잠금 (Optimistic Locking)

```sql
-- version 컬럼 사용
CREATE TABLE products (
  id SERIAL PRIMARY KEY,
  name VARCHAR(255),
  price NUMERIC(10, 2),
  version INT DEFAULT 0
);

-- 업데이트 시 version 확인
UPDATE products
SET
  price = 60000,
  version = version + 1
WHERE id = 10
  AND version = 5;  -- 읽었을 때의 version

-- 결과 확인
GET DIAGNOSTICS updated_rows = ROW_COUNT;
IF updated_rows = 0 THEN
  RAISE EXCEPTION 'Concurrent update detected';
END IF;
```

### 패턴 2: 비관적 잠금 (Pessimistic Locking)

```sql
-- 읽을 때부터 락
BEGIN;
SELECT * FROM products WHERE id = 10 FOR UPDATE;
-- 즉시 락 획득

UPDATE products SET price = 60000 WHERE id = 10;
COMMIT;
```

### 패턴 3: 대기 큐

```sql
-- 작업 큐
CREATE TABLE job_queue (
  id SERIAL PRIMARY KEY,
  job_type VARCHAR(50),
  payload JSONB,
  status VARCHAR(20) DEFAULT 'pending',
  worker_id VARCHAR(50),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  started_at TIMESTAMPTZ,
  completed_at TIMESTAMPTZ
);

-- 워커: 작업 가져가기
CREATE OR REPLACE FUNCTION acquire_job(p_worker_id VARCHAR)
RETURNS TABLE(job_id INT, job_type VARCHAR, payload JSONB) AS $$
BEGIN
  RETURN QUERY
  UPDATE job_queue
  SET
    status = 'processing',
    worker_id = p_worker_id,
    started_at = NOW()
  WHERE id = (
    SELECT id FROM job_queue
    WHERE status = 'pending'
    ORDER BY created_at
    LIMIT 1
    FOR UPDATE SKIP LOCKED
  )
  RETURNING id, job_type, payload;
END;
$$ LANGUAGE plpgsql;

-- 사용
SELECT * FROM acquire_job('worker-1');
```

### 패턴 4: 재시도 로직

```sql
CREATE OR REPLACE FUNCTION update_with_retry(
  p_id INT,
  p_amount NUMERIC,
  p_max_retries INT DEFAULT 3
) RETURNS VOID AS $$
DECLARE
  retry_count INT := 0;
  success BOOLEAN := false;
BEGIN
  WHILE NOT success AND retry_count < p_max_retries LOOP
    BEGIN
      UPDATE accounts
      SET balance = balance + p_amount
      WHERE id = p_id;

      success := true;
    EXCEPTION
      WHEN serialization_failure OR deadlock_detected THEN
        retry_count := retry_count + 1;
        RAISE NOTICE 'Retry % due to %', retry_count, SQLERRM;
        PERFORM pg_sleep(0.1 * retry_count);  -- 백오프
    END;
  END LOOP;

  IF NOT success THEN
    RAISE EXCEPTION 'Failed after % retries', p_max_retries;
  END IF;
END;
$$ LANGUAGE plpgsql;
```

## 18.11 실전 시나리오

### 시나리오 1: 선착순 이벤트

```sql
-- 선착순 100명 쿠폰 지급
CREATE TABLE coupons (
  id SERIAL PRIMARY KEY,
  user_id INT UNIQUE,
  issued_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE coupon_config (
  id INT PRIMARY KEY DEFAULT 1,
  max_count INT DEFAULT 100,
  issued_count INT DEFAULT 0
);

-- 쿠폰 발급
CREATE OR REPLACE FUNCTION issue_coupon(p_user_id INT)
RETURNS BOOLEAN AS $$
DECLARE
  available BOOLEAN;
BEGIN
  -- 설정 잠금
  SELECT issued_count < max_count INTO available
  FROM coupon_config
  WHERE id = 1
  FOR UPDATE;

  IF NOT available THEN
    RETURN false;
  END IF;

  -- 쿠폰 발급
  INSERT INTO coupons (user_id) VALUES (p_user_id);

  -- 카운트 증가
  UPDATE coupon_config SET issued_count = issued_count + 1 WHERE id = 1;

  RETURN true;
END;
$$ LANGUAGE plpgsql;

-- 사용
BEGIN;
SELECT issue_coupon(1);  -- true/false
COMMIT;
```

### 시나리오 2: 재고 관리

```sql
-- 재고 차감
CREATE OR REPLACE FUNCTION reserve_stock(
  p_product_id INT,
  p_quantity INT
) RETURNS BOOLEAN AS $$
DECLARE
  current_stock INT;
BEGIN
  -- 재고 확인 및 락
  SELECT stock INTO current_stock
  FROM products
  WHERE id = p_product_id
  FOR UPDATE;

  -- 재고 부족
  IF current_stock < p_quantity THEN
    RETURN false;
  END IF;

  -- 재고 차감
  UPDATE products
  SET stock = stock - p_quantity
  WHERE id = p_product_id;

  RETURN true;
END;
$$ LANGUAGE plpgsql;

-- 사용
BEGIN;
IF reserve_stock(10, 5) THEN
  -- 주문 생성
  INSERT INTO orders (product_id, quantity) VALUES (10, 5);
  COMMIT;
ELSE
  ROLLBACK;
  RAISE EXCEPTION 'Out of stock';
END IF;
```

## 핵심 요약

### 락 종류

```
행 락:
- FOR UPDATE: 배타적 락
- FOR SHARE: 공유 락
- NOWAIT: 대기 안 함
- SKIP LOCKED: 잠긴 행 건너뛰기

테이블 락:
- ACCESS SHARE: SELECT
- ROW EXCLUSIVE: INSERT, UPDATE, DELETE
- ACCESS EXCLUSIVE: ALTER TABLE, DROP TABLE

Advisory Lock:
- 애플리케이션 레벨 락
- 유연한 제어
```

### FOR UPDATE 옵션

```sql
FOR UPDATE              -- 배타적 락, 대기
FOR UPDATE NOWAIT       -- 배타적 락, 즉시 에러
FOR UPDATE SKIP LOCKED  -- 잠긴 행 건너뛰기

FOR SHARE               -- 공유 락
FOR SHARE NOWAIT        -- 공유 락, 즉시 에러
```

### 데드락 방지

```
1. 잠금 순서 일관성 (ID 순서대로)
2. 잠금 시간 최소화
3. NOWAIT 사용
4. 타임아웃 설정
```

### 모니터링

```sql
-- 락 확인
SELECT * FROM pg_locks;

-- 대기 중인 락
SELECT * FROM pg_stat_activity WHERE wait_event_type = 'Lock';

-- 프로세스 종료
SELECT pg_terminate_backend(pid);
```
