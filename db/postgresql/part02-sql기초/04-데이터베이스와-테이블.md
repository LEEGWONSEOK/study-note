# Chapter 4. 데이터베이스와 테이블

## 4.1 데이터베이스 생성

### CREATE DATABASE

```sql
-- 기본 생성
CREATE DATABASE mydb;

-- 소유자 지정
CREATE DATABASE mydb OWNER myuser;

-- 인코딩 지정
CREATE DATABASE mydb
  ENCODING 'UTF8'
  LC_COLLATE 'ko_KR.UTF-8'
  LC_CTYPE 'ko_KR.UTF-8';

-- 템플릿 사용
CREATE DATABASE mydb TEMPLATE template0;

-- 전체 옵션
CREATE DATABASE mydb
  OWNER myuser
  ENCODING 'UTF8'
  LC_COLLATE 'ko_KR.UTF-8'
  LC_CTYPE 'ko_KR.UTF-8'
  TEMPLATE template0
  CONNECTION LIMIT 100;
```

### 데이터베이스 조회

```sql
-- 데이터베이스 목록
\l
-- 또는
SELECT datname FROM pg_database;

-- 상세 정보
SELECT
  datname AS "Database",
  pg_size_pretty(pg_database_size(datname)) AS "Size",
  datcollate AS "Collation",
  datctype AS "Ctype"
FROM pg_database
ORDER BY pg_database_size(datname) DESC;
```

### 데이터베이스 삭제

```sql
-- 삭제
DROP DATABASE mydb;

-- 존재하면 삭제
DROP DATABASE IF EXISTS mydb;

-- ⚠️ 주의: 연결 중인 DB는 삭제 불가
-- 다른 DB로 전환 후 삭제
\c postgres
DROP DATABASE mydb;
```

## 4.2 스키마

### 스키마란?

```
스키마 = 네임스페이스

비유: 아파트 동
┌─────────────────────────────┐
│ Database (아파트 단지)      │
│                             │
│ ├─ Schema: public (101동)  │
│ │  ├─ Table: users          │
│ │  └─ Table: posts          │
│ │                           │
│ └─ Schema: admin (102동)   │
│    └─ Table: logs           │
└─────────────────────────────┘

용도:
- 논리적 그룹화
- 권한 분리
- 이름 충돌 방지
```

### 스키마 생성

```sql
-- 기본 생성
CREATE SCHEMA myschema;

-- 소유자 지정
CREATE SCHEMA myschema AUTHORIZATION myuser;

-- 존재하면 무시
CREATE SCHEMA IF NOT EXISTS myschema;
```

### 스키마 사용

```sql
-- 1. 스키마 포함하여 테이블 생성
CREATE TABLE myschema.users (
  id SERIAL PRIMARY KEY,
  name VARCHAR(100)
);

-- 2. search_path 설정
SET search_path TO myschema, public;

-- 이제 스키마 없이 접근 가능
SELECT * FROM users;  -- myschema.users

-- 3. search_path 확인
SHOW search_path;
-- myschema, public

-- 4. 기본값으로 복원
SET search_path TO public;
```

### 스키마 조회 및 삭제

```sql
-- 스키마 목록
\dn

-- 상세 정보
SELECT schema_name
FROM information_schema.schemata;

-- 스키마 삭제 (빈 스키마)
DROP SCHEMA myschema;

-- 객체 포함 삭제
DROP SCHEMA myschema CASCADE;
```

## 4.3 테이블 생성 (CREATE TABLE)

### 기본 문법

```sql
CREATE TABLE table_name (
  column_name data_type constraints,
  ...
);
```

### 간단한 예제

```sql
-- 사용자 테이블
CREATE TABLE users (
  id SERIAL PRIMARY KEY,
  email VARCHAR(255) NOT NULL UNIQUE,
  name VARCHAR(100) NOT NULL,
  age INT,
  created_at TIMESTAMP DEFAULT NOW()
);
```

### 컬럼 제약조건

```sql
CREATE TABLE products (
  -- PRIMARY KEY: 기본 키 (NOT NULL + UNIQUE)
  id SERIAL PRIMARY KEY,

  -- NOT NULL: 필수 값
  name VARCHAR(200) NOT NULL,

  -- UNIQUE: 중복 불가
  sku VARCHAR(50) UNIQUE,

  -- DEFAULT: 기본값
  stock INT DEFAULT 0,

  -- CHECK: 조건 검사
  price DECIMAL(10,2) CHECK (price >= 0),

  -- REFERENCES: 외래 키
  category_id INT REFERENCES categories(id),

  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);
```

### 복합 제약조건

```sql
CREATE TABLE orders (
  id SERIAL PRIMARY KEY,
  user_id INT NOT NULL,
  product_id INT NOT NULL,
  quantity INT NOT NULL CHECK (quantity > 0),
  order_date DATE DEFAULT CURRENT_DATE,

  -- 테이블 레벨 제약조건
  UNIQUE (user_id, product_id, order_date),
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE RESTRICT,
  CHECK (quantity <= 100)  -- 한 번에 100개까지만
);
```

## 4.4 SERIAL과 IDENTITY

### SERIAL (레거시)

```sql
-- SERIAL은 SEQUENCE 자동 생성
CREATE TABLE users (
  id SERIAL PRIMARY KEY,  -- INT + SEQUENCE + NOT NULL
  name VARCHAR(100)
);

-- 실제로는 이렇게 생성됨
CREATE SEQUENCE users_id_seq;
CREATE TABLE users (
  id INT NOT NULL DEFAULT nextval('users_id_seq') PRIMARY KEY,
  name VARCHAR(100)
);
```

### IDENTITY (PostgreSQL 10+, 권장)

```sql
-- GENERATED ALWAYS
CREATE TABLE users (
  id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  name VARCHAR(100)
);

-- 수동 삽입 불가
INSERT INTO users (id, name) VALUES (1, 'Alice');
-- ERROR: cannot insert into column "id"

-- GENERATED BY DEFAULT
CREATE TABLE posts (
  id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  title VARCHAR(200)
);

-- 수동 삽입 가능
INSERT INTO posts (id, title) VALUES (100, 'Manual ID');
INSERT INTO posts (title) VALUES ('Auto ID');  -- 101

-- IDENTITY 옵션
CREATE TABLE logs (
  id INT GENERATED ALWAYS AS IDENTITY (
    START WITH 1000
    INCREMENT BY 1
    MINVALUE 1000
    MAXVALUE 999999
    CACHE 20
  ) PRIMARY KEY,
  message TEXT
);
```

## 4.5 테이블 조회

### 기본 조회

```sql
-- 테이블 목록
\dt

-- 특정 스키마
\dt public.*

-- 모든 스키마
\dt *.*

-- 테이블 구조
\d users

-- 상세 정보
\d+ users
```

### SQL로 조회

```sql
-- 테이블 목록
SELECT tablename
FROM pg_tables
WHERE schemaname = 'public';

-- 컬럼 정보
SELECT
  column_name,
  data_type,
  character_maximum_length,
  is_nullable,
  column_default
FROM information_schema.columns
WHERE table_name = 'users'
ORDER BY ordinal_position;

-- 테이블 크기
SELECT
  tablename,
  pg_size_pretty(pg_total_relation_size(schemaname||'.'||tablename)) AS size
FROM pg_tables
WHERE schemaname = 'public'
ORDER BY pg_total_relation_size(schemaname||'.'||tablename) DESC;
```

## 4.6 테이블 수정 (ALTER TABLE)

### 컬럼 추가

```sql
-- 컬럼 추가
ALTER TABLE users ADD COLUMN phone VARCHAR(20);

-- 기본값과 함께
ALTER TABLE users ADD COLUMN status VARCHAR(20) DEFAULT 'active';

-- NOT NULL 추가 (기존 데이터에 기본값 필요)
ALTER TABLE users ADD COLUMN country VARCHAR(50) DEFAULT 'Korea' NOT NULL;

-- 여러 컬럼 추가
ALTER TABLE users
  ADD COLUMN birth_date DATE,
  ADD COLUMN gender CHAR(1);
```

### 컬럼 삭제

```sql
-- 컬럼 삭제
ALTER TABLE users DROP COLUMN phone;

-- CASCADE: 의존 객체도 삭제
ALTER TABLE users DROP COLUMN email CASCADE;

-- 존재하면 삭제
ALTER TABLE users DROP COLUMN IF EXISTS phone;
```

### 컬럼 수정

```sql
-- 데이터 타입 변경
ALTER TABLE users ALTER COLUMN age TYPE BIGINT;

-- 타입 변환 지정
ALTER TABLE users ALTER COLUMN age TYPE VARCHAR(10) USING age::VARCHAR;

-- NOT NULL 추가
ALTER TABLE users ALTER COLUMN email SET NOT NULL;

-- NOT NULL 제거
ALTER TABLE users ALTER COLUMN phone DROP NOT NULL;

-- 기본값 설정
ALTER TABLE users ALTER COLUMN status SET DEFAULT 'active';

-- 기본값 제거
ALTER TABLE users ALTER COLUMN status DROP DEFAULT;

-- 컬럼 이름 변경
ALTER TABLE users RENAME COLUMN name TO full_name;
```

### 제약조건 추가/삭제

```sql
-- UNIQUE 추가
ALTER TABLE users ADD CONSTRAINT users_email_unique UNIQUE (email);

-- CHECK 추가
ALTER TABLE users ADD CONSTRAINT users_age_check CHECK (age >= 0);

-- FOREIGN KEY 추가
ALTER TABLE orders ADD CONSTRAINT orders_user_fk
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE;

-- 제약조건 삭제
ALTER TABLE users DROP CONSTRAINT users_email_unique;

-- 제약조건 조회
\d+ users
```

### 테이블 이름 변경

```sql
-- 테이블 이름 변경
ALTER TABLE users RENAME TO customers;

-- 확인
\dt
```

## 4.7 테이블 삭제

### DROP TABLE

```sql
-- 테이블 삭제
DROP TABLE users;

-- 존재하면 삭제
DROP TABLE IF EXISTS users;

-- CASCADE: 의존 객체도 삭제
DROP TABLE users CASCADE;

-- 여러 테이블 삭제
DROP TABLE IF EXISTS users, posts, comments;
```

### TRUNCATE (데이터만 삭제)

```sql
-- 모든 데이터 삭제 (테이블 구조는 유지)
TRUNCATE TABLE users;

-- RESTART IDENTITY: SERIAL 초기화
TRUNCATE TABLE users RESTART IDENTITY;

-- CASCADE: 참조 테이블도 TRUNCATE
TRUNCATE TABLE users CASCADE;

-- 여러 테이블
TRUNCATE TABLE users, posts, comments RESTART IDENTITY CASCADE;

-- DELETE vs TRUNCATE
-- DELETE: 느림, 롤백 가능, WHERE 가능
-- TRUNCATE: 빠름, 롤백 불가, 전체 삭제만
```

## 4.8 임시 테이블

### TEMPORARY TABLE

```sql
-- 세션 종료 시 자동 삭제
CREATE TEMPORARY TABLE temp_users (
  id SERIAL PRIMARY KEY,
  name VARCHAR(100)
);

-- 약어: TEMP
CREATE TEMP TABLE temp_logs (
  message TEXT,
  created_at TIMESTAMP DEFAULT NOW()
);

-- 데이터 삽입
INSERT INTO temp_users (name) VALUES ('Alice'), ('Bob');

-- 조회
SELECT * FROM temp_users;

-- 세션 종료 후 자동 삭제됨
\q
psql
SELECT * FROM temp_users;  -- ERROR: relation does not exist
```

### 용도

```sql
-- 1. 복잡한 쿼리의 중간 결과
CREATE TEMP TABLE user_stats AS
SELECT
  user_id,
  COUNT(*) AS post_count,
  MAX(created_at) AS last_post
FROM posts
GROUP BY user_id;

SELECT * FROM user_stats WHERE post_count > 10;

-- 2. 데이터 가공
CREATE TEMP TABLE clean_data AS
SELECT
  TRIM(name) AS name,
  LOWER(email) AS email
FROM raw_data
WHERE email IS NOT NULL;

-- 3. 성능 최적화
CREATE TEMP TABLE filtered_users AS
SELECT * FROM users WHERE country = 'Korea';

CREATE INDEX ON filtered_users(email);

SELECT * FROM filtered_users WHERE email LIKE '%@naver.com';
```

## 4.9 테이블 복사

### CREATE TABLE AS

```sql
-- 구조 + 데이터 복사
CREATE TABLE users_backup AS
SELECT * FROM users;

-- 구조만 복사 (데이터 없음)
CREATE TABLE users_template AS
SELECT * FROM users WHERE 1=0;

-- 특정 컬럼만
CREATE TABLE user_emails AS
SELECT id, email FROM users;

-- 조건부 복사
CREATE TABLE active_users AS
SELECT * FROM users WHERE status = 'active';
```

### LIKE (구조만 복사)

```sql
-- 구조만 복사 (제약조건 포함)
CREATE TABLE users_new (LIKE users INCLUDING ALL);

-- 제약조건 제외
CREATE TABLE users_simple (LIKE users);

-- 인덱스 포함
CREATE TABLE users_with_indexes (LIKE users INCLUDING INDEXES);

-- 데이터 복사는 별도로
INSERT INTO users_new SELECT * FROM users;
```

## 4.10 실전 예제

### 블로그 데이터베이스 설계

```sql
-- 1. 데이터베이스 생성
CREATE DATABASE blog_db
  ENCODING 'UTF8'
  LC_COLLATE 'ko_KR.UTF-8'
  LC_CTYPE 'ko_KR.UTF-8';

\c blog_db

-- 2. 사용자 테이블
CREATE TABLE users (
  id SERIAL PRIMARY KEY,
  email VARCHAR(255) UNIQUE NOT NULL,
  username VARCHAR(50) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  full_name VARCHAR(100),
  bio TEXT,
  avatar_url VARCHAR(500),
  is_active BOOLEAN DEFAULT true,
  is_admin BOOLEAN DEFAULT false,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),

  CONSTRAINT username_length CHECK (LENGTH(username) >= 3)
);

-- 3. 카테고리 테이블
CREATE TABLE categories (
  id SERIAL PRIMARY KEY,
  name VARCHAR(100) UNIQUE NOT NULL,
  slug VARCHAR(100) UNIQUE NOT NULL,
  description TEXT,
  created_at TIMESTAMP DEFAULT NOW()
);

-- 4. 게시글 테이블
CREATE TABLE posts (
  id SERIAL PRIMARY KEY,
  user_id INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  category_id INT REFERENCES categories(id) ON DELETE SET NULL,
  title VARCHAR(255) NOT NULL,
  slug VARCHAR(255) UNIQUE NOT NULL,
  content TEXT NOT NULL,
  excerpt TEXT,
  featured_image VARCHAR(500),
  view_count INT DEFAULT 0,
  is_published BOOLEAN DEFAULT false,
  published_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),

  CONSTRAINT title_not_empty CHECK (LENGTH(TRIM(title)) > 0)
);

-- 5. 댓글 테이블
CREATE TABLE comments (
  id SERIAL PRIMARY KEY,
  post_id INT NOT NULL REFERENCES posts(id) ON DELETE CASCADE,
  user_id INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  parent_id INT REFERENCES comments(id) ON DELETE CASCADE,
  content TEXT NOT NULL,
  is_approved BOOLEAN DEFAULT false,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),

  CONSTRAINT content_not_empty CHECK (LENGTH(TRIM(content)) > 0)
);

-- 6. 태그 테이블
CREATE TABLE tags (
  id SERIAL PRIMARY KEY,
  name VARCHAR(50) UNIQUE NOT NULL,
  slug VARCHAR(50) UNIQUE NOT NULL,
  created_at TIMESTAMP DEFAULT NOW()
);

-- 7. 게시글-태그 연결 테이블 (다대다)
CREATE TABLE post_tags (
  post_id INT REFERENCES posts(id) ON DELETE CASCADE,
  tag_id INT REFERENCES tags(id) ON DELETE CASCADE,
  created_at TIMESTAMP DEFAULT NOW(),

  PRIMARY KEY (post_id, tag_id)
);

-- 8. 좋아요 테이블
CREATE TABLE likes (
  user_id INT REFERENCES users(id) ON DELETE CASCADE,
  post_id INT REFERENCES posts(id) ON DELETE CASCADE,
  created_at TIMESTAMP DEFAULT NOW(),

  PRIMARY KEY (user_id, post_id)
);
```

### 샘플 데이터 삽입

```sql
-- 사용자
INSERT INTO users (email, username, password_hash, full_name) VALUES
  ('alice@example.com', 'alice', 'hash1', 'Alice Kim'),
  ('bob@example.com', 'bob', 'hash2', 'Bob Lee');

-- 카테고리
INSERT INTO categories (name, slug, description) VALUES
  ('프로그래밍', 'programming', '프로그래밍 관련 글'),
  ('데이터베이스', 'database', 'DB 관련 글');

-- 게시글
INSERT INTO posts (user_id, category_id, title, slug, content, is_published, published_at)
VALUES
  (1, 1, 'PostgreSQL 시작하기', 'postgresql-intro', '내용...', true, NOW()),
  (2, 2, 'SQL 최적화 팁', 'sql-optimization', '내용...', true, NOW());

-- 태그
INSERT INTO tags (name, slug) VALUES
  ('PostgreSQL', 'postgresql'),
  ('SQL', 'sql'),
  ('성능', 'performance');

-- 게시글-태그 연결
INSERT INTO post_tags (post_id, tag_id) VALUES
  (1, 1), (1, 2),
  (2, 2), (2, 3);

-- 댓글
INSERT INTO comments (post_id, user_id, content, is_approved) VALUES
  (1, 2, '좋은 글 감사합니다!', true);

-- 좋아요
INSERT INTO likes (user_id, post_id) VALUES
  (2, 1),
  (1, 2);
```

### 확인 쿼리

```sql
-- 테이블 목록
\dt

-- 관계 확인
\d posts

-- 게시글과 작성자
SELECT
  p.title,
  u.username,
  c.name AS category
FROM posts p
JOIN users u ON p.user_id = u.id
LEFT JOIN categories c ON p.category_id = c.id;

-- 게시글과 태그
SELECT
  p.title,
  STRING_AGG(t.name, ', ') AS tags
FROM posts p
LEFT JOIN post_tags pt ON p.id = pt.post_id
LEFT JOIN tags t ON pt.tag_id = t.id
GROUP BY p.id, p.title;
```
