# Chapter 7. 조건과 필터링

## 7.1 WHERE 절

### 기본 비교 연산자

```sql
-- = (같음)
SELECT * FROM users WHERE age = 25;

-- != 또는 <> (같지 않음)
SELECT * FROM users WHERE age != 25;
SELECT * FROM users WHERE age <> 25;

-- > (크다)
SELECT * FROM users WHERE age > 30;

-- >= (크거나 같다)
SELECT * FROM users WHERE age >= 30;

-- < (작다)
SELECT * FROM users WHERE age < 20;

-- <= (작거나 같다)
SELECT * FROM users WHERE age <= 20;
```

---

## 7.2 논리 연산자

### AND

```sql
-- 모든 조건 만족
SELECT * FROM users
WHERE age >= 20 AND age <= 30;

-- 여러 조건
SELECT * FROM users
WHERE age > 20
  AND country = 'Korea'
  AND is_active = true;
```

### OR

```sql
-- 하나라도 만족
SELECT * FROM users
WHERE age < 20 OR age > 60;

-- 여러 조건
SELECT * FROM users
WHERE country = 'Korea'
   OR country = 'Japan'
   OR country = 'China';
```

### NOT

```sql
-- 부정
SELECT * FROM users
WHERE NOT age = 25;
-- age != 25와 동일

SELECT * FROM users
WHERE NOT (age < 20 OR age > 60);
-- 20 <= age <= 60과 동일
```

### 조합

```sql
-- AND와 OR 조합 (괄호 주의!)
SELECT * FROM users
WHERE (country = 'Korea' OR country = 'Japan')
  AND age > 20;

-- 우선순위: NOT > AND > OR
SELECT * FROM users
WHERE country = 'Korea'
   OR country = 'Japan' AND age > 20;
-- (Japan AND age > 20) OR Korea

-- 명확하게 괄호 사용 권장
SELECT * FROM users
WHERE country = 'Korea'
   OR (country = 'Japan' AND age > 20);
```

## 7.3 범위 조건

### BETWEEN

```sql
-- 범위 (양끝 포함)
SELECT * FROM users
WHERE age BETWEEN 20 AND 30;
-- age >= 20 AND age <= 30

-- 날짜 범위
SELECT * FROM posts
WHERE created_at BETWEEN '2024-01-01' AND '2024-01-31';

-- NOT BETWEEN
SELECT * FROM users
WHERE age NOT BETWEEN 20 AND 30;
-- age < 20 OR age > 30
```

### IN

```sql
-- 여러 값 중 하나
SELECT * FROM users
WHERE country IN ('Korea', 'Japan', 'China');
-- country = 'Korea' OR country = 'Japan' OR country = 'China'

-- 숫자
SELECT * FROM users
WHERE id IN (1, 5, 10, 15);

-- 서브쿼리
SELECT * FROM posts
WHERE user_id IN (
  SELECT id FROM users WHERE country = 'Korea'
);

-- NOT IN
SELECT * FROM users
WHERE country NOT IN ('Korea', 'Japan');
```

## 7.4 NULL 처리

### IS NULL / IS NOT NULL

```sql
-- NULL 확인
SELECT * FROM users WHERE age IS NULL;

-- NULL이 아닌 값
SELECT * FROM users WHERE age IS NOT NULL;

-- ❌ 잘못된 사용
SELECT * FROM users WHERE age = NULL;  -- 결과 없음!
SELECT * FROM users WHERE age != NULL;  -- 결과 없음!

-- ✅ 올바른 사용
SELECT * FROM users WHERE age IS NULL;
SELECT * FROM users WHERE age IS NOT NULL;
```

### COALESCE (NULL 기본값)

```sql
-- NULL이면 기본값 반환
SELECT
  name,
  COALESCE(age, 0) AS age,
  COALESCE(country, 'Unknown') AS country
FROM users;

-- 여러 값 중 첫 번째 NULL 아닌 값
SELECT COALESCE(NULL, NULL, 'default', 'other');
-- 'default'

-- 실전 예제
SELECT
  name,
  COALESCE(phone, email, 'No contact') AS contact
FROM users;
```

### NULLIF

```sql
-- 두 값이 같으면 NULL 반환
SELECT NULLIF(10, 10);  -- NULL
SELECT NULLIF(10, 20);  -- 10

-- 0으로 나누기 방지
SELECT
  total_sales / NULLIF(order_count, 0) AS avg_sales
FROM stats;
-- order_count가 0이면 NULL (에러 방지)
```

## 7.5 문자열 조건

### LIKE

```sql
-- 패턴 매칭
-- % : 0개 이상의 문자
-- _ : 정확히 1개의 문자

-- 시작
SELECT * FROM users WHERE email LIKE 'admin%';
-- admin@..., administrator@...

-- 끝
SELECT * FROM users WHERE email LIKE '%@gmail.com';

-- 포함
SELECT * FROM users WHERE name LIKE '%kim%';
-- 김철수, alice kim, ...

-- 정확히 5글자
SELECT * FROM users WHERE name LIKE '_____';

-- 조합
SELECT * FROM users WHERE email LIKE '%@%.com';

-- ILIKE (대소문자 무시)
SELECT * FROM users WHERE email ILIKE 'ADMIN%';
-- admin@..., Admin@..., ADMIN@...

-- NOT LIKE
SELECT * FROM users WHERE email NOT LIKE '%@gmail.com';
```

### 정규표현식

```sql
-- ~ : 정규표현식 매칭
SELECT * FROM users WHERE email ~ '^[a-z]+@[a-z]+\.com$';

-- ~* : 대소문자 무시
SELECT * FROM users WHERE email ~* '^ADMIN';

-- !~ : 매칭 안 됨
SELECT * FROM users WHERE email !~ '@gmail\.com$';

-- 실전 예제
-- 한글만
SELECT * FROM users WHERE name ~ '^[가-힣]+$';

-- 영문+숫자만
SELECT * FROM users WHERE username ~ '^[a-zA-Z0-9]+$';

-- 이메일 검증
SELECT * FROM users
WHERE email ~ '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$';
```

### 문자열 함수

```sql
-- LOWER / UPPER
SELECT * FROM users WHERE LOWER(email) = 'admin@example.com';

-- TRIM
SELECT * FROM users WHERE TRIM(name) = 'Alice';

-- LENGTH
SELECT * FROM users WHERE LENGTH(name) > 10;

-- SUBSTRING
SELECT * FROM users
WHERE SUBSTRING(email FROM '@(.*)') = 'gmail.com';

-- POSITION
SELECT * FROM users
WHERE POSITION('@' IN email) > 0;
```

## 7.6 날짜/시간 조건

### 날짜 비교

```sql
-- 특정 날짜
SELECT * FROM posts
WHERE created_at::DATE = '2024-01-15';

-- 날짜 범위
SELECT * FROM posts
WHERE created_at >= '2024-01-01'
  AND created_at < '2024-02-01';

-- 오늘
SELECT * FROM posts
WHERE created_at::DATE = CURRENT_DATE;

-- 어제
SELECT * FROM posts
WHERE created_at::DATE = CURRENT_DATE - 1;

-- 최근 7일
SELECT * FROM posts
WHERE created_at >= CURRENT_DATE - INTERVAL '7 days';

-- 최근 30일
SELECT * FROM posts
WHERE created_at >= NOW() - INTERVAL '30 days';
```

### 날짜 추출

```sql
-- 년도
SELECT * FROM posts
WHERE EXTRACT(YEAR FROM created_at) = 2024;

-- 월
SELECT * FROM posts
WHERE EXTRACT(MONTH FROM created_at) = 1;

-- 요일 (0=일요일, 6=토요일)
SELECT * FROM posts
WHERE EXTRACT(DOW FROM created_at) = 0;  -- 일요일

-- 분기
SELECT * FROM sales
WHERE EXTRACT(QUARTER FROM sale_date) = 1;  -- 1분기

-- DATE_TRUNC (시작일로 잘라내기)
SELECT * FROM posts
WHERE DATE_TRUNC('month', created_at) = '2024-01-01';
-- 2024년 1월

SELECT * FROM posts
WHERE DATE_TRUNC('day', created_at) = CURRENT_DATE;
-- 오늘
```

### 시간 조건

```sql
-- 특정 시간 이후
SELECT * FROM logs
WHERE created_at > '2024-01-15 14:00:00';

-- 특정 시간 범위
SELECT * FROM logs
WHERE EXTRACT(HOUR FROM created_at) BETWEEN 9 AND 18;
-- 9시~18시

-- 업무 시간만
SELECT * FROM logs
WHERE EXTRACT(HOUR FROM created_at) >= 9
  AND EXTRACT(HOUR FROM created_at) < 18
  AND EXTRACT(DOW FROM created_at) BETWEEN 1 AND 5;
-- 평일 9-18시
```

## 7.7 배열 조건

### 배열 연산자

```sql
-- 배열에 값 포함
SELECT * FROM articles
WHERE 'postgresql' = ANY(tags);

-- 모든 값 포함
SELECT * FROM articles
WHERE tags @> ARRAY['postgresql', 'database'];

-- 겹치는 값 있음
SELECT * FROM articles
WHERE tags && ARRAY['postgresql', 'mysql'];

-- 배열 길이
SELECT * FROM articles
WHERE array_length(tags, 1) > 3;
```

## 7.8 JSON 조건

### JSONB 쿼리

```sql
-- JSON 필드 값
SELECT * FROM logs
WHERE data->>'action' = 'login';

-- JSON 필드 존재
SELECT * FROM logs
WHERE data ? 'user_id';

-- JSON 값 포함
SELECT * FROM logs
WHERE data @> '{"action": "login"}';

-- JSON 배열 포함
SELECT * FROM logs
WHERE data->'tags' @> '["urgent"]';

-- 숫자 비교
SELECT * FROM logs
WHERE (data->>'user_id')::INT > 100;
```

## 7.9 CASE 문

### 기본 CASE

```sql
-- 단순 CASE
SELECT
  name,
  age,
  CASE
    WHEN age < 20 THEN '10대'
    WHEN age < 30 THEN '20대'
    WHEN age < 40 THEN '30대'
    ELSE '40대 이상'
  END AS age_group
FROM users;

-- WHERE에서 사용
SELECT * FROM users
WHERE CASE
  WHEN country = 'Korea' THEN age >= 19
  WHEN country = 'USA' THEN age >= 21
  ELSE age >= 18
END;
```

### 실전 예제

```sql
-- 등급 계산
SELECT
  name,
  score,
  CASE
    WHEN score >= 90 THEN 'A'
    WHEN score >= 80 THEN 'B'
    WHEN score >= 70 THEN 'C'
    WHEN score >= 60 THEN 'D'
    ELSE 'F'
  END AS grade
FROM students;

-- 가격 구간
SELECT
  product_name,
  price,
  CASE
    WHEN price < 10000 THEN '저가'
    WHEN price < 50000 THEN '중가'
    ELSE '고가'
  END AS price_range
FROM products;
```

## 7.10 복합 조건 예제

### 사용자 검색

```sql
-- 다양한 조건 조합
SELECT * FROM users
WHERE
  -- 나이 범위
  age BETWEEN 20 AND 40
  -- 국가
  AND country IN ('Korea', 'Japan')
  -- 이메일 도메인
  AND email LIKE '%@gmail.com'
  -- 활성 사용자
  AND is_active = true
  -- 최근 가입
  AND created_at >= NOW() - INTERVAL '30 days'
ORDER BY created_at DESC;
```

### 게시글 필터링

```sql
SELECT * FROM posts
WHERE
  -- 게시됨
  is_published = true
  -- 카테고리
  AND category_id IN (1, 2, 3)
  -- 제목 검색
  AND (
    title ILIKE '%postgresql%'
    OR content ILIKE '%postgresql%'
  )
  -- 최근 1년
  AND created_at >= NOW() - INTERVAL '1 year'
  -- 조회수 많음
  AND view_count > 100
ORDER BY view_count DESC, created_at DESC
LIMIT 20;
```

### 주문 검색

```sql
SELECT * FROM orders
WHERE
  -- 날짜 범위
  order_date BETWEEN '2024-01-01' AND '2024-01-31'
  -- 금액 범위
  AND total_amount >= 100000
  -- 상태
  AND status IN ('pending', 'processing')
  -- 배송지
  AND shipping_country = 'Korea'
  -- 특정 사용자 제외
  AND user_id NOT IN (
    SELECT id FROM blacklist_users
  )
ORDER BY order_date DESC;
```

## 7.11 성능 최적화

### 인덱스 활용

```sql
-- ❌ 느림: 함수 사용
SELECT * FROM users WHERE LOWER(email) = 'admin@example.com';

-- ✅ 빠름: 원본 값 사용
SELECT * FROM users WHERE email = 'admin@example.com';

-- 함수 사용 필요 시 인덱스 생성
CREATE INDEX idx_users_email_lower ON users (LOWER(email));

-- ❌ 느림: LIKE 앞에 %
SELECT * FROM users WHERE email LIKE '%@gmail.com';

-- ✅ 빠름: LIKE 뒤에만 %
SELECT * FROM users WHERE email LIKE 'admin%';
```

### OR vs IN

```sql
-- ❌ 느림: OR 많음
SELECT * FROM users
WHERE id = 1 OR id = 2 OR id = 3 OR ... OR id = 100;

-- ✅ 빠름: IN 사용
SELECT * FROM users
WHERE id IN (1, 2, 3, ..., 100);
```

### NOT IN vs NOT EXISTS

```sql
-- ❌ 느림: NOT IN (NULL 문제)
SELECT * FROM users
WHERE id NOT IN (SELECT user_id FROM blacklist);

-- ✅ 빠름: NOT EXISTS
SELECT * FROM users u
WHERE NOT EXISTS (
  SELECT 1 FROM blacklist b WHERE b.user_id = u.id
);
```
