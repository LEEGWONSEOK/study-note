# Chapter 19. 멀티프로세싱 (Multiprocessing)

> **Java 개발자를 위한 노트**: Java의 `ProcessBuilder`나 `Runtime.exec()`와 달리, Python의 `multiprocessing`은 동일한 Python 코드를 여러 프로세스에서 쉽게 실행할 수 있습니다. GIL에서 자유롭기 때문에 CPU 집약적 작업에 최적입니다.

## 1. 멀티프로세싱이란?

### 1.1 프로세스 vs 스레드

**비유**:
- **스레드**: 한 회사(프로세스) 안에서 여러 직원이 화장실(GIL) 열쇠를 공유하며 일하는 것
- **프로세스**: 여러 독립된 회사가 각자 화장실 열쇠를 가지고 완전히 독립적으로 운영되는 것

```python
"""
┌─────────────────────────────────────┐
│         멀티스레딩                    │
│  ┌─────────────────────────────┐   │
│  │   Process (공유 메모리)      │   │
│  │  ┌────┐ ┌────┐ ┌────┐      │   │
│  │  │T1  │ │T2  │ │T3  │      │   │
│  │  │ 🔒 ← GIL → 🔒 │      │   │
│  │  └────┘ └────┘ └────┘      │   │
│  └─────────────────────────────┘   │
└─────────────────────────────────────┘

┌─────────────────────────────────────┐
│        멀티프로세싱                   │
│  ┌──────┐  ┌──────┐  ┌──────┐     │
│  │ P1   │  │ P2   │  │ P3   │     │
│  │ 🔒  │  │ 🔒  │  │ 🔒  │     │
│  │독립  │  │독립  │  │독립  │     │
│  │메모리│  │메모리│  │메모리│     │
│  └──────┘  └──────┘  └──────┘     │
└─────────────────────────────────────┘
"""
```

### Java vs Python

**Java**:
```java
// Java - ProcessBuilder로 외부 프로세스 실행
ProcessBuilder pb = new ProcessBuilder("python", "script.py");
Process process = pb.start();

// 또는 Thread를 사용 (GIL 없음)
ExecutorService executor = Executors.newFixedThreadPool(4);
executor.submit(() -> cpuIntensiveTask());
```

**Python**:
```python
# Python - 같은 코드를 여러 프로세스에서 실행
from multiprocessing import Process

def cpu_intensive_task():
    # GIL 없이 완전히 병렬 실행
    result = sum(i*i for i in range(10000000))
    return result

if __name__ == '__main__':
    p1 = Process(target=cpu_intensive_task)
    p2 = Process(target=cpu_intensive_task)

    p1.start()
    p2.start()

    p1.join()
    p2.join()
```

## 2. Process 기본

### 2.1 프로세스 생성

```python
from multiprocessing import Process
import os
import time

def worker(name, duration):
    """작업자 프로세스"""
    print(f"🚀 [{name}] 시작 (PID: {os.getpid()})")
    time.sleep(duration)
    print(f"✅ [{name}] 완료")

if __name__ == '__main__':
    print(f"🏢 메인 프로세스 PID: {os.getpid()}")

    # 프로세스 생성
    p1 = Process(target=worker, args=('Worker-1', 2))
    p2 = Process(target=worker, args=('Worker-2', 1))

    # 시작
    p1.start()
    p2.start()

    # 완료 대기
    p1.join()
    p2.join()

    print("🎉 모든 작업 완료")

# 출력:
# 🏢 메인 프로세스 PID: 12345
# 🚀 [Worker-1] 시작 (PID: 12346)
# 🚀 [Worker-2] 시작 (PID: 12347)
# ✅ [Worker-2] 완료
# ✅ [Worker-1] 완료
# 🎉 모든 작업 완료
```

### 2.2 반환값 받기

**비유**: 프로세스는 다른 회사이므로, 우편(Queue, Pipe)으로 결과를 주고받아야 합니다.

```python
from multiprocessing import Process, Queue

def calculate_sum(numbers, result_queue):
    """숫자 합계 계산"""
    total = sum(numbers)
    result_queue.put(total)
    print(f"📊 프로세스 {os.getpid()}: {total} 계산 완료")

if __name__ == '__main__':
    # Queue 생성 (프로세스 간 통신)
    result_queue = Queue()

    # 데이터 분할
    data1 = range(0, 5000000)
    data2 = range(5000000, 10000000)

    # 프로세스 생성
    p1 = Process(target=calculate_sum, args=(data1, result_queue))
    p2 = Process(target=calculate_sum, args=(data2, result_queue))

    start = time.time()

    p1.start()
    p2.start()

    # 결과 수집
    result1 = result_queue.get()
    result2 = result_queue.get()

    p1.join()
    p2.join()

    total = result1 + result2
    elapsed = time.time() - start

    print(f"✅ 총합: {total:,}")
    print(f"⏱️  소요 시간: {elapsed:.2f}초")
```

## 3. Process Pool

**비유**: `Pool`은 여러 프로세스를 미리 만들어두고 재활용하는 **프로세스 풀장**입니다.

### 3.1 기본 사용법

```python
from multiprocessing import Pool
import time

def square(n):
    """제곱 계산"""
    print(f"📐 {n}^2 계산 중... (PID: {os.getpid()})")
    time.sleep(0.1)
    return n * n

if __name__ == '__main__':
    numbers = [1, 2, 3, 4, 5, 6, 7, 8]

    # 4개의 프로세스로 풀 생성
    with Pool(processes=4) as pool:
        results = pool.map(square, numbers)

    print(f"결과: {results}")

# 출력:
# 📐 1^2 계산 중... (PID: 12348)
# 📐 2^2 계산 중... (PID: 12349)
# 📐 3^2 계산 중... (PID: 12350)
# 📐 4^2 계산 중... (PID: 12351)
# 📐 5^2 계산 중... (PID: 12348)  # 재활용!
# ...
# 결과: [1, 4, 9, 16, 25, 36, 49, 64]
```

### 3.2 Pool 주요 메서드

```python
from multiprocessing import Pool

def process_task(x):
    return x * x

if __name__ == '__main__':
    with Pool(4) as pool:
        # 1. map - 순서 보장, 블로킹
        results = pool.map(process_task, [1, 2, 3, 4])
        print(f"map: {results}")

        # 2. map_async - 비블로킹
        async_result = pool.map_async(process_task, [5, 6, 7, 8])
        print("다른 작업 수행 가능...")
        results = async_result.get()  # 결과 대기
        print(f"map_async: {results}")

        # 3. apply - 단일 작업, 블로킹
        result = pool.apply(process_task, (10,))
        print(f"apply: {result}")

        # 4. apply_async - 단일 작업, 비블로킹
        async_result = pool.apply_async(process_task, (20,))
        result = async_result.get()
        print(f"apply_async: {result}")
```

### 3.3 실전 예제 - 이미지 처리

```python
from multiprocessing import Pool
from PIL import Image
import os

def resize_image(image_path):
    """이미지 리사이징"""
    try:
        print(f"🖼️  처리 중: {image_path}")

        # 이미지 열기
        img = Image.open(image_path)

        # 썸네일 생성
        img.thumbnail((200, 200))

        # 저장
        base_name = os.path.basename(image_path)
        output_path = f"thumbnails/{base_name}"
        img.save(output_path)

        return f"✅ {base_name} 완료"

    except Exception as e:
        return f"❌ {image_path} 실패: {e}"

def batch_resize_images(image_folder, num_processes=4):
    """배치 이미지 리사이징"""
    # 이미지 파일 목록
    image_files = [
        os.path.join(image_folder, f)
        for f in os.listdir(image_folder)
        if f.endswith(('.jpg', '.png', '.jpeg'))
    ]

    print(f"📁 총 {len(image_files)}개 이미지 처리 시작")

    # 출력 폴더 생성
    os.makedirs('thumbnails', exist_ok=True)

    # 멀티프로세싱으로 처리
    with Pool(processes=num_processes) as pool:
        results = pool.map(resize_image, image_files)

    # 결과 출력
    for result in results:
        print(result)

if __name__ == '__main__':
    batch_resize_images('images', num_processes=4)
```

## 4. 프로세스 간 통신 (IPC)

### 4.1 Queue - 안전한 데이터 전달

**비유**: Queue는 프로세스 간 **우체통**입니다. 한쪽에서 편지를 넣으면, 다른 쪽에서 꺼냅니다.

```python
from multiprocessing import Process, Queue
import time

def producer(queue):
    """생산자 - 데이터를 Queue에 넣음"""
    for i in range(5):
        item = f"Item-{i}"
        queue.put(item)
        print(f"📤 생산: {item}")
        time.sleep(0.5)

    queue.put(None)  # 종료 신호

def consumer(queue):
    """소비자 - Queue에서 데이터를 꺼냄"""
    while True:
        item = queue.get()
        if item is None:
            print("🛑 종료 신호 수신")
            break

        print(f"📥 소비: {item}")
        time.sleep(1)

if __name__ == '__main__':
    q = Queue()

    p1 = Process(target=producer, args=(q,))
    p2 = Process(target=consumer, args=(q,))

    p1.start()
    p2.start()

    p1.join()
    p2.join()
```

### 4.2 Pipe - 양방향 통신

**비유**: Pipe는 두 프로세스 사이의 **전화선**입니다.

```python
from multiprocessing import Process, Pipe

def worker(conn):
    """작업자 프로세스"""
    # 메시지 수신
    msg = conn.recv()
    print(f"📞 받음: {msg}")

    # 응답 전송
    conn.send(f"Echo: {msg}")

    conn.close()

if __name__ == '__main__':
    # Pipe 생성 (양방향)
    parent_conn, child_conn = Pipe()

    p = Process(target=worker, args=(child_conn,))
    p.start()

    # 메시지 전송
    parent_conn.send("Hello from parent!")

    # 응답 수신
    response = parent_conn.recv()
    print(f"📨 응답: {response}")

    p.join()
```

### 4.3 Manager - 공유 자료구조

**비유**: Manager는 프로세스들이 공유하는 **공용 화이트보드**입니다.

```python
from multiprocessing import Process, Manager

def worker(shared_dict, shared_list, worker_id):
    """공유 자료구조 사용"""
    # 딕셔너리에 추가
    shared_dict[worker_id] = f"Worker-{worker_id}"

    # 리스트에 추가
    shared_list.append(worker_id)

    print(f"✍️  Worker-{worker_id} 작업 완료")

if __name__ == '__main__':
    with Manager() as manager:
        # 공유 자료구조 생성
        shared_dict = manager.dict()
        shared_list = manager.list()

        # 프로세스 생성
        processes = []
        for i in range(5):
            p = Process(target=worker, args=(shared_dict, shared_list, i))
            processes.append(p)
            p.start()

        # 완료 대기
        for p in processes:
            p.join()

        print(f"\n📋 공유 딕셔너리: {dict(shared_dict)}")
        print(f"📋 공유 리스트: {list(shared_list)}")

# 출력:
# ✍️  Worker-0 작업 완료
# ✍️  Worker-1 작업 완료
# ✍️  Worker-2 작업 완료
# ✍️  Worker-3 작업 완료
# ✍️  Worker-4 작업 완료
#
# 📋 공유 딕셔너리: {0: 'Worker-0', 1: 'Worker-1', ...}
# 📋 공유 리스트: [0, 1, 2, 3, 4]
```

## 5. Lock과 동기화

**비유**: Lock은 프로세스 간 **신호등**입니다.

```python
from multiprocessing import Process, Lock, Value
import time

def increment_counter(lock, counter, worker_id):
    """카운터 증가 (Lock 사용)"""
    for _ in range(100):
        with lock:  # Lock 획득
            current = counter.value
            time.sleep(0.001)  # 경쟁 조건 시뮬레이션
            counter.value = current + 1

    print(f"✅ Worker-{worker_id} 완료")

if __name__ == '__main__':
    # 공유 변수 생성
    lock = Lock()
    counter = Value('i', 0)  # 'i' = integer

    # 프로세스 생성
    processes = []
    for i in range(5):
        p = Process(target=increment_counter, args=(lock, counter, i))
        processes.append(p)
        p.start()

    # 완료 대기
    for p in processes:
        p.join()

    print(f"\n🎯 최종 카운터: {counter.value}")
    print(f"📊 예상값: {5 * 100}")

# Lock 없으면: 최종 카운터가 500보다 작을 수 있음
# Lock 있으면: 항상 500
```

## 6. 실전 예제 - CPU 집약적 작업

### 6.1 소수 찾기

```python
from multiprocessing import Pool
import time
import math

def is_prime(n):
    """소수 판별"""
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False

    for i in range(3, int(math.sqrt(n)) + 1, 2):
        if n % i == 0:
            return False
    return True

def find_primes_sequential(start, end):
    """순차 처리"""
    return [n for n in range(start, end) if is_prime(n)]

def find_primes_parallel(start, end, num_processes=4):
    """병렬 처리"""
    # 범위 분할
    chunk_size = (end - start) // num_processes
    ranges = [
        (start + i * chunk_size, start + (i + 1) * chunk_size)
        for i in range(num_processes)
    ]
    ranges[-1] = (ranges[-1][0], end)  # 마지막 범위 조정

    # 병렬 처리
    with Pool(processes=num_processes) as pool:
        results = pool.starmap(find_primes_in_range, ranges)

    # 결과 병합
    return [p for sublist in results for p in sublist]

def find_primes_in_range(start, end):
    """범위 내 소수 찾기"""
    return [n for n in range(start, end) if is_prime(n)]

if __name__ == '__main__':
    start_num = 1
    end_num = 100000

    # 순차 처리
    print("⏱️  순차 처리 시작...")
    start = time.time()
    primes_seq = find_primes_sequential(start_num, end_num)
    time_seq = time.time() - start
    print(f"✅ 소수 개수: {len(primes_seq)}, 시간: {time_seq:.2f}초")

    # 병렬 처리
    print("\n⚡ 병렬 처리 시작...")
    start = time.time()
    primes_par = find_primes_parallel(start_num, end_num, num_processes=4)
    time_par = time.time() - start
    print(f"✅ 소수 개수: {len(primes_par)}, 시간: {time_par:.2f}초")

    # 성능 비교
    speedup = time_seq / time_par
    print(f"\n🚀 속도 향상: {speedup:.2f}x")

# 예상 출력:
# ⏱️  순차 처리 시작...
# ✅ 소수 개수: 9592, 시간: 2.34초
#
# ⚡ 병렬 처리 시작...
# ✅ 소수 개수: 9592, 시간: 0.68초
#
# 🚀 속도 향상: 3.44x
```

### 6.2 데이터 분석 - CSV 처리

```python
from multiprocessing import Pool
import csv
import os

def analyze_csv_chunk(file_path, start_row, end_row):
    """CSV 청크 분석"""
    totals = {'sales': 0, 'count': 0}

    with open(file_path, 'r') as f:
        reader = csv.DictReader(f)

        # 시작 위치로 이동
        for _ in range(start_row):
            next(reader)

        # 데이터 처리
        for i, row in enumerate(reader):
            if i >= (end_row - start_row):
                break

            totals['sales'] += float(row['sales'])
            totals['count'] += 1

    return totals

def analyze_large_csv_parallel(file_path, num_processes=4):
    """대용량 CSV 병렬 분석"""
    # 총 라인 수 계산
    with open(file_path, 'r') as f:
        total_rows = sum(1 for _ in f) - 1  # 헤더 제외

    # 청크 분할
    chunk_size = total_rows // num_processes
    tasks = [
        (file_path, i * chunk_size, (i + 1) * chunk_size)
        for i in range(num_processes)
    ]
    tasks[-1] = (file_path, tasks[-1][1], total_rows)

    # 병렬 처리
    with Pool(processes=num_processes) as pool:
        results = pool.starmap(analyze_csv_chunk, tasks)

    # 결과 집계
    total_sales = sum(r['sales'] for r in results)
    total_count = sum(r['count'] for r in results)

    return {
        'total_sales': total_sales,
        'count': total_count,
        'average': total_sales / total_count if total_count > 0 else 0
    }

if __name__ == '__main__':
    result = analyze_large_csv_parallel('sales_data.csv', num_processes=4)

    print(f"📊 분석 결과:")
    print(f"   총 매출: ${result['total_sales']:,.2f}")
    print(f"   거래 수: {result['count']:,}")
    print(f"   평균 매출: ${result['average']:,.2f}")
```

### 7.2 비교 예제

```python
from concurrent.futures import ThreadPoolExecutor, ProcessPoolExecutor
import time
import requests

# CPU 집약적 작업
def cpu_bound_task(n):
    """CPU 집약적 - 프로세스 적합"""
    return sum(i*i for i in range(n))

# I/O 집약적 작업
def io_bound_task(url):
    """I/O 집약적 - 스레드 적합"""
    response = requests.get(url)
    return len(response.content)

if __name__ == '__main__':
    # CPU 집약적 - 프로세스가 더 빠름
    numbers = [5000000] * 4

    print("🧮 CPU 집약적 작업:")

    # 스레드로 실행
    start = time.time()
    with ThreadPoolExecutor(max_workers=4) as executor:
        results = list(executor.map(cpu_bound_task, numbers))
    print(f"  스레드: {time.time() - start:.2f}초")

    # 프로세스로 실행
    start = time.time()
    with ProcessPoolExecutor(max_workers=4) as executor:
        results = list(executor.map(cpu_bound_task, numbers))
    print(f"  프로세스: {time.time() - start:.2f}초 ⚡")

    # I/O 집약적 - 스레드가 더 효율적
    urls = ['https://www.google.com'] * 4

    print("\n📡 I/O 집약적 작업:")

    # 스레드로 실행
    start = time.time()
    with ThreadPoolExecutor(max_workers=4) as executor:
        results = list(executor.map(io_bound_task, urls))
    print(f"  스레드: {time.time() - start:.2f}초 ⚡")

    # 프로세스로 실행
    start = time.time()
    with ProcessPoolExecutor(max_workers=4) as executor:
        results = list(executor.map(io_bound_task, urls))
    print(f"  프로세스: {time.time() - start:.2f}초")
```

```python
from multiprocessing import Pool
import os

def search_in_file(args):
    """파일에서 키워드 검색"""
    file_path, keyword = args
    matches = []

    try:
        with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
            for line_num, line in enumerate(f, 1):
                if keyword.lower() in line.lower():
                    matches.append({
                        'file': file_path,
                        'line': line_num,
                        'text': line.strip()
                    })
    except Exception as e:
        pass

    return matches

def get_all_files(directories):
    """모든 파일 경로 수집"""
    all_files = []
    for directory in directories:
        for root, _, files in os.walk(directory):
            for file in files:
                if file.endswith(('.txt', '.py', '.md')):
                    all_files.append(os.path.join(root, file))
    return all_files

def parallel_file_search(directories, keyword, num_processes=4):
    """병렬 파일 검색"""
    # 모든 파일 수집
    all_files = get_all_files(directories)
    print(f"📁 검색할 파일: {len(all_files)}개")

    # 병렬 검색
    tasks = [(f, keyword) for f in all_files]

    with Pool(processes=num_processes) as pool:
        results = pool.map(search_in_file, tasks)

    # 결과 병합
    all_matches = []
    for matches in results:
        all_matches.extend(matches)

    return all_matches
```

### 문제 3: 행렬 곱셈
대용량 행렬의 곱셈을 멀티프로세싱으로 최적화하세요.

```python
from multiprocessing import Pool
import numpy as np
import time

def multiply_row(args):
    """한 행의 곱셈 계산"""
    row_idx, row, matrix_b = args
    result_row = np.dot(row, matrix_b)
    return row_idx, result_row

def parallel_matrix_multiply(matrix_a, matrix_b, num_processes=4):
    """병렬 행렬 곱셈"""
    rows, cols = matrix_a.shape

    # 각 행마다 작업 생성
    tasks = [
        (i, matrix_a[i], matrix_b)
        for i in range(rows)
    ]

    # 병렬 처리
    with Pool(processes=num_processes) as pool:
        results = pool.map(multiply_row, tasks)

    # 결과 조합
    result_matrix = np.zeros((rows, matrix_b.shape[1]))
    for row_idx, row_result in results:
        result_matrix[row_idx] = row_result

    return result_matrix

if __name__ == '__main__':
    # 대용량 행렬 생성
    size = 1000
    A = np.random.rand(size, size)
    B = np.random.rand(size, size)

    # NumPy 기본 곱셈
    print("🔢 NumPy 기본 곱셈:")
    start = time.time()
    result_numpy = np.dot(A, B)
    time_numpy = time.time() - start
    print(f"   시간: {time_numpy:.2f}초")

    # 병렬 곱셈
    print("\n⚡ 병렬 곱셈:")
    start = time.time()
    result_parallel = parallel_matrix_multiply(A, B, num_processes=4)
    time_parallel = time.time() - start
    print(f"   시간: {time_parallel:.2f}초")

    # 결과 검증
    if np.allclose(result_numpy, result_parallel):
        print("\n✅ 결과 일치!")
    else:
        print("\n❌ 결과 불일치!")

    # 참고: NumPy는 내부적으로 최적화되어 있어서
    # 일반적으로 순수 Python 멀티프로세싱보다 빠릅니다
```

## 핵심 요약

1. **멀티프로세싱 개념**
   - 프로세스는 독립된 메모리 공간을 가진 실행 단위
   - GIL에서 자유로워 CPU 집약적 작업에 최적
   - 프로세스 간 통신(IPC)이 필요

2. **주요 도구**
   ```python
   # Process
   p = Process(target=func, args=(arg,))
   p.start()
   p.join()

   # Pool
   with Pool(4) as pool:
       results = pool.map(func, iterable)

   # Queue (IPC)
   q = Queue()
   q.put(item)
   item = q.get()
   ```

3. **언제 사용하나**
   - ✅ CPU 집약적 작업 (계산, 데이터 분석, 이미지 처리)
   - ✅ 대용량 데이터 병렬 처리
   - ❌ I/O 집약적 작업 (스레드나 asyncio가 더 효율적)

4. **주의사항**
   - `if __name__ == '__main__'` 필수 (특히 Windows)
   - 공유 상태 최소화
   - 적절한 프로세스 수 선택 (보통 CPU 코어 수)

5. **Java와 비교**
   - Java: 스레드가 진정한 병렬 실행 (GIL 없음)
   - Python: 프로세스로 진정한 병렬 실행
