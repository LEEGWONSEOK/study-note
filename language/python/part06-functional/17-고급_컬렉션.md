## 17.2 Counter - 개수 세기

> 🔢 **비유**: Counter는 **투표 집계기**
> - 각 항목의 등장 횟수 자동 카운트

```python
from collections import Counter

# 기본 사용
words = ['apple', 'banana', 'apple', 'cherry', 'banana', 'apple']
counter = Counter(words)

print(counter)
# Counter({'apple': 3, 'banana': 2, 'cherry': 1})

# 개별 접근
print(counter['apple'])   # 3
print(counter['grape'])   # 0 (없어도 에러 안 남!)

# 가장 흔한 요소
print(counter.most_common(2))
# [('apple', 3), ('banana', 2)]
```

---

### Counter 연산

```python
from collections import Counter

# 카운터 생성
c1 = Counter(['a', 'b', 'c', 'a', 'b', 'b'])
c2 = Counter(['a', 'b', 'd', 'b', 'd', 'd'])

print(c1)  # Counter({'b': 3, 'a': 2, 'c': 1})
print(c2)  # Counter({'d': 3, 'b': 2, 'a': 1})

# 덧셈
print(c1 + c2)
# Counter({'b': 5, 'a': 3, 'd': 3, 'c': 1})

# 뺄셈 (음수는 제거)
print(c1 - c2)
# Counter({'b': 1, 'a': 1, 'c': 1})

# 교집합 (최소값)
print(c1 & c2)
# Counter({'a': 1, 'b': 2})

# 합집합 (최대값)
print(c1 | c2)
# Counter({'b': 3, 'd': 3, 'a': 2, 'c': 1})
```

### 실전 예제: 단어 빈도 분석

```python
from collections import Counter
import re

def analyze_text(text):
    """텍스트 분석"""
    # 단어 추출 (소문자 변환)
    words = re.findall(r'\w+', text.lower())

    # 빈도수 계산
    word_count = Counter(words)

    # 통계
    total_words = len(words)
    unique_words = len(word_count)
    most_common = word_count.most_common(5)

    return {
        'total': total_words,
        'unique': unique_words,
        'most_common': most_common,
        'counter': word_count
    }

# 사용
text = """
Python is a great programming language.
Python is easy to learn and Python is powerful.
Many developers love Python.
"""

stats = analyze_text(text)
print(f"전체 단어: {stats['total']}")
print(f"고유 단어: {stats['unique']}")
print("\n가장 많이 사용된 단어:")
for word, count in stats['most_common']:
    print(f"  {word}: {count}회")

# 출력:
# 전체 단어: 18
# 고유 단어: 14
#
# 가장 많이 사용된 단어:
#   python: 4회
#   is: 3회
#   and: 1회
#   a: 1회
#   great: 1회
```

## 17.3 defaultdict - 기본값 딕셔너리

> 🎁 **비유**: defaultdict는 **자동 선물 포장**
> - 키가 없으면 자동으로 기본값 생성

```python
from collections import defaultdict

# 일반 dict (KeyError)
normal_dict = {}
# normal_dict['key'] += 1  # KeyError!

# defaultdict (자동 생성)
dd = defaultdict(int)  # 기본값: 0
dd['key'] += 1
print(dd['key'])  # 1

# 리스트 기본값
dd_list = defaultdict(list)
dd_list['fruits'].append('apple')
dd_list['fruits'].append('banana')
print(dd_list['fruits'])  # ['apple', 'banana']

# 집합 기본값
dd_set = defaultdict(set)
dd_set['numbers'].add(1)
dd_set['numbers'].add(2)
dd_set['numbers'].add(1)  # 중복 무시
print(dd_set['numbers'])  # {1, 2}
```

### 실전 예제: 그룹화

```python
from collections import defaultdict

# 학생들을 학년별로 그룹화
students = [
    {'name': 'Alice', 'grade': 1},
    {'name': 'Bob', 'grade': 2},
    {'name': 'Charlie', 'grade': 1},
    {'name': 'David', 'grade': 2},
    {'name': 'Eve', 'grade': 1}
]

# defaultdict 사용
groups = defaultdict(list)
for student in students:
    groups[student['grade']].append(student['name'])

print(dict(groups))
# {1: ['Alice', 'Charlie', 'Eve'], 2: ['Bob', 'David']}

# vs 일반 dict (번거로움)
groups_normal = {}
for student in students:
    grade = student['grade']
    if grade not in groups_normal:
        groups_normal[grade] = []
    groups_normal[grade].append(student['name'])
```

## 17.4 deque - 양방향 큐

> 🚂 **비유**: deque는 **양방향 기차**
> - 앞뒤 모두에서 빠르게 추가/제거

```python
from collections import deque

# 생성
d = deque([1, 2, 3])

# 오른쪽 추가/제거
d.append(4)
print(d)  # deque([1, 2, 3, 4])

d.pop()
print(d)  # deque([1, 2, 3])

# 왼쪽 추가/제거
d.appendleft(0)
print(d)  # deque([0, 1, 2, 3])

d.popleft()
print(d)  # deque([1, 2, 3])

# 확장
d.extend([4, 5])
print(d)  # deque([1, 2, 3, 4, 5])

d.extendleft([0, -1])  # 역순으로 추가됨!
print(d)  # deque([-1, 0, 1, 2, 3, 4, 5])

# 회전
d.rotate(2)  # 오른쪽으로 2칸
print(d)  # deque([4, 5, -1, 0, 1, 2, 3])

d.rotate(-3)  # 왼쪽으로 3칸
print(d)  # deque([0, 1, 2, 3, 4, 5, -1])
```

### 성능 비교: list vs deque

```python
import time
from collections import deque

# 리스트
lst = []
start = time.time()
for i in range(100000):
    lst.insert(0, i)  # 앞에 삽입 (느림!)
list_time = time.time() - start

# deque
dq = deque()
start = time.time()
for i in range(100000):
    dq.appendleft(i)  # 앞에 삽입 (빠름!)
deque_time = time.time() - start

print(f"List: {list_time:.4f}초")
print(f"Deque: {deque_time:.4f}초")
print(f"Deque가 {list_time/deque_time:.0f}배 빠름")

# List: 3.2145초
# Deque: 0.0089초
# Deque가 361배 빠름
```

### 실전 예제: 최근 N개 유지

```python
from collections import deque

class RecentItems:
    """최근 N개 항목만 유지"""

    def __init__(self, max_size=5):
        self.items = deque(maxlen=max_size)

    def add(self, item):
        """항목 추가 (오래된 것 자동 제거)"""
        self.items.append(item)

    def get_recent(self):
        """최근 항목들"""
        return list(self.items)

# 사용
history = RecentItems(max_size=3)

for i in range(1, 6):
    history.add(f"Page {i}")
    print(f"방문: Page {i} → 최근: {history.get_recent()}")

# 방문: Page 1 → 최근: ['Page 1']
# 방문: Page 2 → 최근: ['Page 1', 'Page 2']
# 방문: Page 3 → 최근: ['Page 1', 'Page 2', 'Page 3']
# 방문: Page 4 → 최근: ['Page 2', 'Page 3', 'Page 4']
# 방문: Page 5 → 최근: ['Page 3', 'Page 4', 'Page 5']
```

## 17.5 namedtuple - 이름있는 튜플

> 🏷️ **비유**: namedtuple은 **라벨 붙은 상자**
> - 인덱스 대신 이름으로 접근
> - 불변 (immutable)

```python
from collections import namedtuple

# 정의
Point = namedtuple('Point', ['x', 'y'])

# 생성
p = Point(10, 20)

# 접근 (인덱스 또는 이름)
print(p[0])    # 10
print(p.x)     # 10
print(p.y)     # 20

# 언패킹
x, y = p
print(x, y)    # 10 20

# 불변
# p.x = 15  # AttributeError!

# 새 객체 생성 (_replace)
p2 = p._replace(x=15)
print(p2)  # Point(x=15, y=20)

# 딕셔너리로 변환
print(p._asdict())  # {'x': 10, 'y': 20}

# 필드 확인
print(Point._fields)  # ('x', 'y')
```

## 17.6 OrderedDict - 순서 유지 딕셔너리

> 📋 **비유**: OrderedDict는 **순서 있는 명부**
> - 삽입 순서 유지 (Python 3.7+부터는 일반 dict도 순서 유지)

```python
from collections import OrderedDict

# OrderedDict
od = OrderedDict()
od['c'] = 3
od['a'] = 1
od['b'] = 2

print(od)
# OrderedDict([('c', 3), ('a', 1), ('b', 2)])

# 순서 변경
od.move_to_end('a')  # 끝으로 이동
print(od)
# OrderedDict([('c', 3), ('b', 2), ('a', 1)])

od.move_to_end('c', last=False)  # 처음으로 이동
print(od)
# OrderedDict([('c', 3), ('b', 2), ('a', 1)])

# FIFO 삭제
od.popitem(last=False)  # 첫 번째 제거
print(od)
# OrderedDict([('b', 2), ('a', 1)])
```

> 💡 **참고**: Python 3.7+에서는 일반 dict도 순서를 유지하지만, OrderedDict는 순서 관련 메서드 제공

## 17.7 ChainMap - 연결된 딕셔너리

> 🔗 **비유**: ChainMap은 **딕셔너리 체인**
> - 여러 딕셔너리를 하나처럼 사용
> - 우선순위 순서대로 검색

```python
from collections import ChainMap

# 여러 딕셔너리
defaults = {'color': 'blue', 'size': 'medium'}
user_prefs = {'color': 'red'}
cli_args = {'size': 'large'}

# ChainMap 생성 (왼쪽이 우선)
config = ChainMap(cli_args, user_prefs, defaults)

print(config['color'])  # red (user_prefs)
print(config['size'])   # large (cli_args)

# 모든 딕셔너리 확인
print(config.maps)
# [{'size': 'large'}, {'color': 'red'}, {'color': 'blue', 'size': 'medium'}]

# 새 자식 추가
config = config.new_child({'color': 'green'})
print(config['color'])  # green (새 자식이 최우선)
```

### 실전 예제: 설정 우선순위

```python
from collections import ChainMap
import os

# 설정 우선순위: CLI > 환경변수 > 기본값
def get_config(cli_args=None):
    """설정 가져오기"""
    # 기본값
    defaults = {
        'host': 'localhost',
        'port': 8000,
        'debug': False
    }

    # 환경변수
    env_vars = {
        k.lower().replace('app_', ''): v
        for k, v in os.environ.items()
        if k.startswith('APP_')
    }

    # CLI 인수
    cli = cli_args or {}

    # ChainMap (우선순위 순)
    return ChainMap(cli, env_vars, defaults)

# 사용
config = get_config({'port': 9000})

print(f"Host: {config['host']}")    # localhost (defaults)
print(f"Port: {config['port']}")    # 9000 (cli)
print(f"Debug: {config['debug']}")  # False (defaults)
```

## 17.8 실전 예제

### 예제 1: LRU 캐시 구현

```python
from collections import OrderedDict

class LRUCache:
    """Least Recently Used 캐시"""

    def __init__(self, capacity):
        self.cache = OrderedDict()
        self.capacity = capacity

    def get(self, key):
        """값 가져오기"""
        if key not in self.cache:
            return None

        # 최근 사용으로 이동
        self.cache.move_to_end(key)
        return self.cache[key]

    def put(self, key, value):
        """값 저장"""
        if key in self.cache:
            # 업데이트
            self.cache.move_to_end(key)
        else:
            # 새 항목
            if len(self.cache) >= self.capacity:
                # 가장 오래된 것 제거
                self.cache.popitem(last=False)

        self.cache[key] = value

    def __str__(self):
        return str(dict(self.cache))

# 사용
cache = LRUCache(3)

cache.put('a', 1)
cache.put('b', 2)
cache.put('c', 3)
print(cache)  # {'a': 1, 'b': 2, 'c': 3}

cache.get('a')  # 'a' 최근 사용
cache.put('d', 4)  # 'b' 제거 (가장 오래됨)
print(cache)  # {'c': 3, 'a': 1, 'd': 4}
```

### 예제 2: 이동 평균 계산

```python
from collections import deque

class MovingAverage:
    """이동 평균"""

    def __init__(self, window_size):
        self.window = deque(maxlen=window_size)
        self.sum = 0

    def add(self, value):
        """값 추가"""
        if len(self.window) == self.window.maxlen:
            # 가장 오래된 값 제거
            self.sum -= self.window[0]

        self.window.append(value)
        self.sum += value

    def average(self):
        """평균 계산"""
        if not self.window:
            return 0
        return self.sum / len(self.window)

# 사용
ma = MovingAverage(3)

values = [10, 20, 30, 40, 50]
for val in values:
    ma.add(val)
    print(f"값: {val}, 평균: {ma.average()}")

# 값: 10, 평균: 10.0
# 값: 20, 평균: 15.0
# 값: 30, 평균: 20.0
# 값: 40, 평균: 30.0
# 값: 50, 평균: 40.0
```

### 예제 3: 문자 빈도 분석기

```python
from collections import Counter, defaultdict

class TextAnalyzer:
    """텍스트 분석기"""

    def __init__(self, text):
        self.text = text.lower()
        self.char_count = Counter(self.text)
        self.word_groups = self._group_by_length()

    def _group_by_length(self):
        """단어 길이별 그룹화"""
        import re
        words = re.findall(r'\w+', self.text)

        groups = defaultdict(list)
        for word in words:
            groups[len(word)].append(word)

        return dict(groups)

    def most_common_chars(self, n=5):
        """가장 많은 문자"""
        return self.char_count.most_common(n)

    def words_by_length(self, length):
        """특정 길이의 단어들"""
        return self.word_groups.get(length, [])

    def statistics(self):
        """통계"""
        return {
            'total_chars': len(self.text),
            'unique_chars': len(self.char_count),
            'total_words': sum(len(words) for words in self.word_groups.values()),
            'word_lengths': sorted(self.word_groups.keys())
        }

# 사용
text = """
Python is a great programming language.
It is easy to learn and powerful.
"""

analyzer = TextAnalyzer(text)

print("통계:")
stats = analyzer.statistics()
for key, value in stats.items():
    print(f"  {key}: {value}")

print("\n가장 많이 사용된 문자:")
for char, count in analyzer.most_common_chars():
    if char.isalpha():
        print(f"  {char}: {count}회")

print("\n6글자 단어:")
print(analyzer.words_by_length(6))
```
