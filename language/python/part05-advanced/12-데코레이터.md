# Chapter 12. 데코레이터 (Decorators)

## 12.1 데코레이터란?

> 🎁 **비유**: 데코레이터는 **선물 포장**입니다.
> - 선물(함수): 본래 기능
> - 포장지(데코레이터): 추가 기능
> - 선물은 그대로, 겉모습만 달라짐

**데코레이터**: 함수나 클래스를 **수정하지 않고** 기능을 추가하는 문법

---

```python
def add_greeting_wrapper(func):
```

### @ 문법 사용

```python
def add_greeting_wrapper(func):

        print("=== 인사 시작 ===")
        result = func(name)
        print("=== 인사 끝 ===")

@add_greeting_wrapper  # greet = add_greeting_wrapper(greet)
```

## 12.3 실용적인 데코레이터

### 1. 시간 측정

```python
import time
from functools import wraps

def timer(func):
    """함수 실행 시간 측정"""
    @wraps(func)  # 메타데이터 보존
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        elapsed = time.time() - start
        print(f"{func.__name__} 실행 시간: {elapsed:.4f}초")
        return result
    return wrapper

@timer
def slow_sum(n):
    """1부터 n까지 합계"""
    total = 0
    for i in range(n):
        total += i
    return total

result = slow_sum(1000000)
print(f"결과: {result}")

# 출력:
# slow_sum 실행 시간: 0.0532초
# 결과: 499999500000
```

```python
from functools import wraps
import datetime

def log_calls(func):
    """함수 호출 로깅"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")

        print(f"  인수: args={args}, kwargs={kwargs}")

        print(f"  반환: {result}")
```

### 3. 캐싱 (메모이제이션)

```python
from functools import wraps

def cache(func):
    """결과 캐싱"""
    cached_results = {}

    @wraps(func)
    def wrapper(*args):
        if args in cached_results:
            print(f"캐시에서 반환: {args}")
            return cached_results[args]

        print(f"계산 중: {args}")
        result = func(*args)
        cached_results[args] = result
        return result

    return wrapper

@cache
def fibonacci(n):
    """피보나치 수열"""
    if n <= 1:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

print(fibonacci(10))
print(fibonacci(10))  # 캐시에서 즉시 반환

# 출력:
# 계산 중: (10,)
# 계산 중: (9,)
# 계산 중: (8,)
# ...
# 55
# 캐시에서 반환: (10,)
# 55
```

```python
# 전역 변수 (실제로는 세션 등에서 관리)
current_user = None

def login_required(func):
    """로그인 필수"""
```

## 12.4 매개변수가 있는 데코레이터

> 🎛️ **비유**: **설정 가능한 포장지**
> - 빨간 리본? 파란 리본?
> - 데코레이터에 옵션 전달

```python
from functools import wraps

def repeat(times):
    """함수를 여러 번 실행"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            results = []
            for _ in range(times):
                result = func(*args, **kwargs)
                results.append(result)
            return results
        return wrapper
    return decorator

@repeat(times=3)
def greet(name):
    return f"Hello, {name}!"

print(greet("Alice"))
# ['Hello, Alice!', 'Hello, Alice!', 'Hello, Alice!']
```

**구조**:
```python
# @repeat(times=3)
# def greet(name):
#     ...

# 실제로는:
# greet = repeat(times=3)(greet)
#       = decorator(greet)
#       = wrapper
```

### 실전 예제: 재시도 데코레이터

```python
import time
from functools import wraps

def retry(max_attempts=3, delay=1):
    """실패 시 재시도"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(1, max_attempts + 1):
                try:
                    print(f"시도 {attempt}/{max_attempts}")
                    result = func(*args, **kwargs)
                    print("성공!")
                    return result
                except Exception as e:
                    print(f"실패: {e}")
                    if attempt < max_attempts:
                        print(f"{delay}초 후 재시도...")
                        time.sleep(delay)
                    else:
                        print("모든 재시도 실패")
                        raise
        return wrapper
    return decorator

@retry(max_attempts=3, delay=1)
def unstable_api_call(success_rate=0.5):
    """불안정한 API (시뮬레이션)"""
    import random
    if random.random() > success_rate:
        raise ConnectionError("API 호출 실패")
    return {"status": "success", "data": "결과"}

# 테스트
try:
    result = unstable_api_call(success_rate=0.3)  # 30% 성공률
    print(f"결과: {result}")
except ConnectionError:
    print("최종 실패")
```

## 12.6 클래스 데코레이터

> 🏭 **비유**: 클래스 데코레이터는 **공장 개조**
> - 클래스 전체를 수정/확장

```python
def singleton(cls):
    """싱글톤 패턴"""
    instances = {}

    def get_instance(*args, **kwargs):
        if cls not in instances:
            instances[cls] = cls(*args, **kwargs)
        return instances[cls]

    return get_instance

@singleton
class DatabaseConnection:
    def __init__(self, host):
        print(f"DB 연결: {host}")
        self.host = host

# 테스트
db1 = DatabaseConnection("localhost")  # DB 연결: localhost
db2 = DatabaseConnection("localhost")  # 출력 없음 (이미 생성됨)

print(db1 is db2)  # True (같은 인스턴스)
```

### 실전 예제: 자동 속성 추가

```python
def add_str_method(cls):
    """__str__ 메서드 자동 생성"""
    def __str__(self):
        attrs = ', '.join(f"{k}={v}" for k, v in self.__dict__.items())
        return f"{cls.__name__}({attrs})"

    cls.__str__ = __str__
    return cls

@add_str_method
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age

@add_str_method
class Product:
    def __init__(self, name, price):
        self.name = name
        self.price = price

# 사용
person = Person("Alice", 25)
print(person)  # Person(name=Alice, age=25)

product = Product("노트북", 1000000)
print(product)  # Product(name=노트북, price=1000000)
```

### @total_ordering - 비교 연산 자동 생성

```python
from functools import total_ordering

@total_ordering
class Student:
    def __init__(self, name, grade):
        self.name = name
        self.grade = grade

    def __eq__(self, other):
        return self.grade == other.grade

    def __lt__(self, other):
        return self.grade < other.grade

    # __le__, __gt__, __ge__는 자동 생성됨!

# 사용
s1 = Student("Alice", 90)
s2 = Student("Bob", 85)

print(s1 > s2)   # True
print(s1 >= s2)  # True
print(s1 <= s2)  # False
```

## 12.8 메서드 데코레이터

### @property

```python
class Temperature:
    def __init__(self, celsius):
        self._celsius = celsius

    @property
    def celsius(self):
        """섭씨 (읽기)"""
        return self._celsius

    @celsius.setter
    def celsius(self, value):
        """섭씨 (쓰기)"""
        if value < -273.15:
            raise ValueError("절대영도 이하 불가")
        self._celsius = value

    @property
    def fahrenheit(self):
        """화씨 (계산)"""
        return self._celsius * 9/5 + 32

# 사용
temp = Temperature(25)
print(temp.celsius)     # 25
print(temp.fahrenheit)  # 77.0

temp.celsius = 100
print(temp.fahrenheit)  # 212.0
```

### @classmethod, @staticmethod

```python
class MathUtils:

    # 인스턴스 메서드
```

```python
class SimpleRouter:
    """간단한 라우팅 시스템"""

    def __init__(self):
        self.routes = {}

    def route(self, path):
        """라우트 등록 데코레이터"""
        def decorator(func):
            self.routes[path] = func
            return func
        return decorator

    def handle(self, path, *args, **kwargs):
        """요청 처리"""
        if path in self.routes:
            return self.routes[path](*args, **kwargs)
        return "404 Not Found"


# 사용
app = SimpleRouter()

@app.route("/")
def home():
    return "Welcome to Home!"

@app.route("/about")
def about():
    return "About Page"

@app.route("/user")
def user_page(user_id):
    return f"User Profile: {user_id}"

# 요청 처리
print(app.handle("/"))           # Welcome to Home!
print(app.handle("/about"))      # About Page
print(app.handle("/user", 123))  # User Profile: 123
print(app.handle("/invalid"))    # 404 Not Found
```

### 예제 2: 유효성 검사 데코레이터

```python
from functools import wraps

def validate_types(**type_checks):
    """타입 검증"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            # 키워드 인수 검증
            for arg_name, expected_type in type_checks.items():
                if arg_name in kwargs:
                    value = kwargs[arg_name]
                    if not isinstance(value, expected_type):
                        raise TypeError(
                            f"{arg_name}은 {expected_type.__name__} 타입이어야 합니다"
                        )
            return func(*args, **kwargs)
        return wrapper
    return decorator

def validate_range(arg_name, min_val=None, max_val=None):
    """범위 검증"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            if arg_name in kwargs:
                value = kwargs[arg_name]
                if min_val is not None and value < min_val:
                    raise ValueError(f"{arg_name}은 {min_val} 이상이어야 합니다")
                if max_val is not None and value > max_val:
                    raise ValueError(f"{arg_name}은 {max_val} 이하여야 합니다")
            return func(*args, **kwargs)
        return wrapper
    return decorator

@validate_types(name=str, age=int, salary=float)
@validate_range('age', min_val=0, max_val=150)
@validate_range('salary', min_val=0)
def create_employee(name, age, salary):
    return {
        "name": name,
        "age": age,
        "salary": salary
    }

# 성공
emp = create_employee(name="Alice", age=25, salary=50000.0)
print(emp)

# 타입 오류
try:
    create_employee(name="Bob", age="thirty", salary=50000.0)
except TypeError as e:
    print(f"타입 오류: {e}")

# 범위 오류
try:
    create_employee(name="Charlie", age=200, salary=50000.0)
except ValueError as e:
    print(f"범위 오류: {e}")
```

### 예제 3: 성능 프로파일러

```python
import time
from functools import wraps

class PerformanceProfiler:
    """성능 측정"""

    def __init__(self):
        self.stats = {}

    def profile(self, func):
        """프로파일링 데코레이터"""
        @wraps(func)
        def wrapper(*args, **kwargs):
            start = time.time()
            result = func(*args, **kwargs)
            elapsed = time.time() - start

            # 통계 기록
            if func.__name__ not in self.stats:
                self.stats[func.__name__] = {
                    'calls': 0,
                    'total_time': 0,
                    'min_time': float('inf'),
                    'max_time': 0
                }

            stats = self.stats[func.__name__]
            stats['calls'] += 1
            stats['total_time'] += elapsed
            stats['min_time'] = min(stats['min_time'], elapsed)
            stats['max_time'] = max(stats['max_time'], elapsed)

            return result
        return wrapper

    def report(self):
        """통계 리포트"""
        print("\n=== 성능 리포트 ===")
        for name, stats in self.stats.items():
            avg_time = stats['total_time'] / stats['calls']
            print(f"\n{name}:")
            print(f"  호출 횟수: {stats['calls']}")
            print(f"  총 시간: {stats['total_time']:.4f}초")
            print(f"  평균 시간: {avg_time:.4f}초")
            print(f"  최소 시간: {stats['min_time']:.4f}초")
            print(f"  최대 시간: {stats['max_time']:.4f}초")


# 사용
profiler = PerformanceProfiler()

@profiler.profile
def process_data(n):

    return sum(range(n))

@profiler.profile
```

## 실전 팁

### 💡 Tip 1: @wraps 항상 사용

```python
from functools import wraps

# ❌ 나쁜 예
def bad_decorator(func):
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

# ✅ 좋은 예
def good_decorator(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper
```

### 💡 Tip 3: 선택적 매개변수

```python
from functools import wraps

def optional_decorator(func=None, *, param=None):
    """매개변수가 선택적인 데코레이터"""
    def decorator(f):
        @wraps(f)
        def wrapper(*args, **kwargs):
            print(f"Param: {param}")
            return f(*args, **kwargs)
        return wrapper

    if func is None:
        # @optional_decorator(param="value")
        return decorator
    else:
        # @optional_decorator
        return decorator(func)

# 두 가지 방식 모두 가능
@optional_decorator
def func1():
    return "No param"

@optional_decorator(param="custom")
def func2():
    return "With param"

func1()  # Param: None
func2()  # Param: custom
```

### 문제 1: 디버그 데코레이터

함수 이름, 인수, 반환값을 출력하는 디버그 데코레이터 작성

<details>
<summary>정답 보기</summary>

```python
from functools import wraps

def debug(func):
    """디버그 정보 출력"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        args_str = ', '.join(repr(a) for a in args)
        kwargs_str = ', '.join(f"{k}={v!r}" for k, v in kwargs.items())
        all_args = ', '.join(filter(None, [args_str, kwargs_str]))

        print(f"호출: {func.__name__}({all_args})")
        result = func(*args, **kwargs)
        print(f"반환: {result!r}")

        return result
    return wrapper

@debug
def add(a, b):
    return a + b

@debug
def greet(name, greeting="Hello"):
    return f"{greeting}, {name}!"

add(10, 5)
# 호출: add(10, 5)
# 반환: 15

greet("Alice", greeting="Hi")
# 호출: greet('Alice', greeting='Hi')
# 반환: 'Hi, Alice!'
```
</details>

### 문제 2: 횟수 제한 데코레이터

함수를 최대 n번까지만 호출 가능하도록 제한

<details>
<summary>정답 보기</summary>

```python
from functools import wraps

def limit_calls(max_calls):
    """호출 횟수 제한"""
    def decorator(func):
        call_count = [0]  # 리스트로 감싸서 변경 가능하게

        @wraps(func)
        def wrapper(*args, **kwargs):
            if call_count[0] >= max_calls:
                raise RuntimeError(
                    f"{func.__name__}은 최대 {max_calls}번만 호출 가능합니다"
                )
            call_count[0] += 1
            return func(*args, **kwargs)

        return wrapper
    return decorator

@limit_calls(3)
def greet(name):
    return f"Hello, {name}!"

print(greet("Alice"))  # 1번
print(greet("Bob"))    # 2번
print(greet("Charlie")) # 3번

try:
    print(greet("David"))  # 4번 - 오류!
except RuntimeError as e:
    print(e)  # greet은 최대 3번만 호출 가능합니다
```
</details>

   ```python
   def decorator(func):
       @wraps(func)
       def wrapper(*args, **kwargs):
           # 전처리
           result = func(*args, **kwargs)
           # 후처리
           return result
       return wrapper
   ```

   ```python
   def decorator(param):
       def inner(func):
           @wraps(func)

               return func(*args, **kwargs)
           return wrapper
       return inner
   ```
