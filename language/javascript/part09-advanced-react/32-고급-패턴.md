## Render Props 패턴

**렌더링 로직을 prop으로 전달**하는 패턴입니다.

### 비유로 이해하기
Render Props는 **맞춤 제작 서비스**와 같습니다:
- 기본 틀(로직)은 제공
- 최종 디자인(UI)은 고객이 결정
- 재사용 가능하면서도 유연함

### 기본 개념

```javascript
// 마우스 위치를 추적하는 컴포넌트
function MouseTracker({ render }) {
    const [position, setPosition] = useState({ x: 0, y: 0 });

    useEffect(() => {
        const handleMove = (e) => {
            setPosition({ x: e.clientX, y: e.clientY });
        };

        window.addEventListener('mousemove', handleMove);
        return () => window.removeEventListener('mousemove', handleMove);
    }, []);

    // 렌더링 로직을 prop으로 받음
    return render(position);
}

// 사용 1: 좌표 표시
function App() {
    return (
        <MouseTracker
            render={({ x, y }) => (
                <h1>마우스 위치: ({x}, {y})</h1>
            )}
        />
    );
}

// 사용 2: 고양이 따라다니게
function App() {
    return (
        <MouseTracker
            render={({ x, y }) => (
                <img
                    src="/cat.png"
                    style={{ position: 'absolute', left: x, top: y }}
                />
            )}
        />
    );
}
```

### Children as Function 패턴

```javascript
// children을 함수로 사용
function DataFetcher({ url, children }) {
    const [data, setData] = useState(null);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState(null);

    useEffect(() => {
        fetch(url)
            .then(res => res.json())
            .then(data => {
                setData(data);
                setLoading(false);
            })
            .catch(err => {
                setError(err);
                setLoading(false);
            });
    }, [url]);

    return children({ data, loading, error });
}

// 사용
function App() {
    return (
        <DataFetcher url="/api/users">
            {({ data, loading, error }) => {
                if (loading) return <div>로딩 중...</div>;
                if (error) return <div>에러: {error.message}</div>;
                return (
                    <ul>
                        {data.map(user => (
                            <li key={user.id}>{user.name}</li>
                        ))}
                    </ul>
                );
            }}
        </DataFetcher>
    );
}
```

// Toggle 컴포넌트
function Toggle({ children }) {
    const [on, setOn] = useState(false);

    const toggle = () => setOn(!on);
    const setToOn = () => setOn(true);
    const setToOff = () => setOn(false);

    return children({
        on,
        toggle,
        setToOn,
        setToOff
    });
}

// 사용 1: 모달
function ModalExample() {
    return (
        <Toggle>
            {({ on, setToOn, setToOff }) => (

                            <button onClick={setToOff}>닫기</button>
                        </div>
                    )}
                </div>
            )}
        </Toggle>
    );
}

## Higher-Order Components (HOC)

**컴포넌트를 받아서 새로운 컴포넌트를 반환**하는 함수입니다.

### 기본 개념

```javascript
// HOC: 컴포넌트를 받아 새 컴포넌트 반환
function withLoading(Component) {
    return function WithLoadingComponent({ isLoading, ...props }) {
        if (isLoading) {
            return <div>로딩 중...</div>;
        }
        return <Component {...props} />;
    };
}

// 원래 컴포넌트
function UserList({ users }) {
    return (
        <ul>
            {users.map(user => (
                <li key={user.id}>{user.name}</li>
            ))}
        </ul>
    );
}

// HOC 적용
const UserListWithLoading = withLoading(UserList);

// 사용
function App() {
    const [users, setUsers] = useState([]);
    const [loading, setLoading] = useState(true);

    return (
        <UserListWithLoading
            isLoading={loading}
            users={users}
        />
    );
}
```

### 실전 HOC 예제

```javascript
// 1. 인증 확인 HOC
function withAuth(Component) {
    return function WithAuthComponent(props) {
        const { user, loading } = useAuth();

        if (loading) {
            return <div>로딩 중...</div>;
        }

        if (!user) {
            return <Navigate to="/login" />;
        }

        return <Component {...props} user={user} />;
    };
}

// 사용
const ProtectedDashboard = withAuth(Dashboard);

function App() {
    return (
        <Routes>
            <Route path="/dashboard" element={<ProtectedDashboard />} />
        </Routes>
    );
}

// 2. 데이터 가져오기 HOC
function withData(Component, url) {
    return function WithDataComponent(props) {
        const [data, setData] = useState(null);
        const [loading, setLoading] = useState(true);
        const [error, setError] = useState(null);

        useEffect(() => {
            fetch(url)
                .then(res => res.json())
                .then(data => {
                    setData(data);
                    setLoading(false);
                })
                .catch(err => {
                    setError(err);
                    setLoading(false);
                });
        }, []);

        return (
            <Component
                {...props}
                data={data}
                loading={loading}
                error={error}
            />
        );
    };
}

// 사용
const UserListWithData = withData(UserList, '/api/users');

// 3. 로깅 HOC
function withLogging(Component) {
    return function WithLoggingComponent(props) {
        useEffect(() => {
            console.log('컴포넌트 마운트됨:', Component.name);
            return () => {
                console.log('컴포넌트 언마운트됨:', Component.name);
            };
        }, []);

        useEffect(() => {
            console.log('Props 변경됨:', props);
        }, [props]);

        return <Component {...props} />;
    };
}

// 4. HOC 조합
const EnhancedComponent = withAuth(
    withLogging(
        withData(UserList, '/api/users')
    )
);
```

## Compound Components 패턴

**여러 컴포넌트가 함께 동작**하는 패턴입니다.

### 비유로 이해하기
Compound Components는 **레고 세트**와 같습니다:
- 각 부품은 독립적이지만
- 함께 사용할 때 완전한 기능
- 조합 방식은 사용자가 결정

### 기본 패턴

```javascript
// Tab 컴포넌트
const TabsContext = createContext();

function Tabs({ children, defaultTab }) {
    const [activeTab, setActiveTab] = useState(defaultTab);

    return (
        <TabsContext.Provider value={{ activeTab, setActiveTab }}>
            <div className="tabs">{children}</div>
        </TabsContext.Provider>
    );
}

function TabList({ children }) {
    return <div className="tab-list">{children}</div>;
}

function Tab({ id, children }) {
    const { activeTab, setActiveTab } = useContext(TabsContext);
    const isActive = activeTab === id;

    return (
        <button
            className={`tab ${isActive ? 'active' : ''}`}
            onClick={() => setActiveTab(id)}
        >
            {children}
        </button>
    );
}

function TabPanels({ children }) {
    return <div className="tab-panels">{children}</div>;
}

function TabPanel({ id, children }) {
    const { activeTab } = useContext(TabsContext);

    if (activeTab !== id) return null;

    return <div className="tab-panel">{children}</div>;
}

// 컴포넌트를 속성으로 연결
Tabs.List = TabList;
Tabs.Tab = Tab;
Tabs.Panels = TabPanels;
Tabs.Panel = TabPanel;

// 사용
function App() {
    return (
        <Tabs defaultTab="home">
            <Tabs.List>
                <Tabs.Tab id="home">홈</Tabs.Tab>
                <Tabs.Tab id="profile">프로필</Tabs.Tab>
                <Tabs.Tab id="settings">설정</Tabs.Tab>
            </Tabs.List>

            <Tabs.Panels>
                <Tabs.Panel id="home">
                    <h2>홈 페이지</h2>
                </Tabs.Panel>
                <Tabs.Panel id="profile">
                    <h2>프로필 페이지</h2>
                </Tabs.Panel>
                <Tabs.Panel id="settings">
                    <h2>설정 페이지</h2>
                </Tabs.Panel>
            </Tabs.Panels>
        </Tabs>
    );
}
```

### 실전 예제: Accordion

```javascript
const AccordionContext = createContext();

function Accordion({ children, allowMultiple = false }) {
    const [openItems, setOpenItems] = useState([]);

    const toggleItem = (id) => {
        if (allowMultiple) {
            setOpenItems(prev =>
                prev.includes(id)
                    ? prev.filter(item => item !== id)
                    : [...prev, id]
            );
        } else {
            setOpenItems(prev =>
                prev.includes(id) ? [] : [id]
            );
        }
    };

    const isOpen = (id) => openItems.includes(id);

    return (
        <AccordionContext.Provider value={{ toggleItem, isOpen }}>
            <div className="accordion">{children}</div>
        </AccordionContext.Provider>
    );
}

function AccordionItem({ id, children }) {
    return <div className="accordion-item">{children}</div>;
}

function AccordionHeader({ id, children }) {
    const { toggleItem, isOpen } = useContext(AccordionContext);

    return (
        <button
            className="accordion-header"
            onClick={() => toggleItem(id)}
        >
            {children}
            <span>{isOpen(id) ? '▲' : '▼'}</span>
        </button>
    );
}

function AccordionPanel({ id, children }) {
    const { isOpen } = useContext(AccordionContext);

    if (!isOpen(id)) return null;

    return <div className="accordion-panel">{children}</div>;
}

Accordion.Item = AccordionItem;
Accordion.Header = AccordionHeader;
Accordion.Panel = AccordionPanel;

// 사용
function FAQ() {
    return (
        <Accordion allowMultiple>
            <Accordion.Item id="1">
                <Accordion.Header id="1">
                    React란 무엇인가요?
                </Accordion.Header>
                <Accordion.Panel id="1">
                    React는 사용자 인터페이스를 구축하기 위한 라이브러리입니다.
                </Accordion.Panel>
            </Accordion.Item>

            <Accordion.Item id="2">
                <Accordion.Header id="2">
                    왜 React를 사용하나요?
                </Accordion.Header>
                <Accordion.Panel id="2">
                    컴포넌트 기반 아키텍처와 가상 DOM으로 효율적입니다.
                </Accordion.Panel>
            </Accordion.Item>
        </Accordion>
    );
}
```

## 제어 vs 비제어 컴포넌트

### 제어 컴포넌트 (Controlled)

**React state가 유일한 진실 공급원**입니다.

```javascript
function ControlledInput() {
    const [value, setValue] = useState('');

    // React가 완전히 제어
    return (
        <input
            value={value}
            onChange={(e) => setValue(e.target.value)}
        />
    );
}

// 제어 컴포넌트의 장점
function ControlledForm() {
    const [email, setEmail] = useState('');

    // 실시간 유효성 검사
    const isValid = /\S+@\S+\.\S+/.test(email);

    // 입력값 변환
    const handleChange = (e) => {
        setEmail(e.target.value.toLowerCase());
    };

    return (
        <div>
            <input
                type="email"
                value={email}
                onChange={handleChange}
            />
            {!isValid && <span>유효한 이메일을 입력하세요</span>}
            <button disabled={!isValid}>제출</button>
        </div>
    );
}
```

### 비제어 컴포넌트 (Uncontrolled)

**DOM이 진실 공급원**입니다.

```javascript
function UncontrolledInput() {
    const inputRef = useRef(null);

    const handleSubmit = (e) => {
        e.preventDefault();
        // 필요할 때만 DOM에서 값 가져오기
        console.log(inputRef.current.value);
    };

    return (
        <form onSubmit={handleSubmit}>
            <input ref={inputRef} defaultValue="초기값" />
            <button type="submit">제출</button>
        </form>
    );
}

// 파일 입력은 항상 비제어
function FileUpload() {
    const fileRef = useRef(null);

    const handleSubmit = (e) => {
        e.preventDefault();
        const files = fileRef.current.files;
        console.log(files);
    };

    return (
        <form onSubmit={handleSubmit}>
            <input type="file" ref={fileRef} />
            <button type="submit">업로드</button>
        </form>
    );
}
```

### 하이브리드 패턴

```javascript
// 제어/비제어를 모두 지원하는 컴포넌트
function FlexibleInput({
    value: controlledValue,
    defaultValue,
    onChange
}) {
    const [internalValue, setInternalValue] = useState(defaultValue || '');

    // 제어 모드인지 확인
    const isControlled = controlledValue !== undefined;
    const value = isControlled ? controlledValue : internalValue;

    const handleChange = (e) => {
        const newValue = e.target.value;

        if (!isControlled) {
            setInternalValue(newValue);
        }

        onChange?.(e);
    };

    return (
        <input
            value={value}
            onChange={handleChange}
        />
    );
}

// 제어 모드로 사용
function ControlledUsage() {
    const [value, setValue] = useState('');

    return (
        <FlexibleInput
            value={value}
            onChange={(e) => setValue(e.target.value)}
        />
    );
}

// 비제어 모드로 사용
function UncontrolledUsage() {
    return (
        <FlexibleInput
            defaultValue="초기값"
            onChange={(e) => console.log(e.target.value)}
        />
    );
}
```
