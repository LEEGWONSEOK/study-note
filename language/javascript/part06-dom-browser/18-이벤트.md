## 이벤트란?

이벤트는 **웹 페이지에서 발생하는 사용자 액션이나 시스템 알림**입니다.

### 비유로 이해하기
이벤트는 **도어벨**과 같습니다:
- 사용자가 버튼을 누름 (이벤트 발생)
- 벨이 울림 (이벤트 전파)
- 누군가 반응함 (이벤트 핸들러 실행)
- 여러 사람이 들을 수 있음 (여러 리스너)

### Java와 비교

```java
// Java Swing: ActionListener
JButton button = new JButton("클릭");
button.addActionListener(new ActionListener() {
    @Override
    public void actionPerformed(ActionEvent e) {
        System.out.println("버튼 클릭됨!");
    }

button.addActionListener(e -> {
    System.out.println("버튼 클릭됨!");
});
```

```javascript

button.addEventListener('click', function(event) {
```

## 이벤트 리스너

### 이벤트 등록 방법

```javascript
const button = document.querySelector('button');

// 1. HTML 속성 (권장하지 않음)
// <button onclick="handleClick()">클릭</button>

// 2. DOM 속성 (하나만 가능)
button.onclick = function() {
    console.log('클릭됨');
};

// 나중에 덮어쓰기됨
button.onclick = function() {
    console.log('다시 클릭됨');
};

// 3. addEventListener (권장)
button.addEventListener('click', function() {
    console.log('첫 번째 리스너');
});

button.addEventListener('click', function() {
    console.log('두 번째 리스너');
}); // 둘 다 실행됨!

// 화살표 함수 사용
button.addEventListener('click', (event) => {
    console.log('화살표 함수', event);
});
```

### 이벤트 리스너 제거

```javascript
const button = document.querySelector('button');

// 이름이 있는 함수 사용
function handleClick(event) {
    console.log('클릭됨');
}

button.addEventListener('click', handleClick);

// 나중에 제거
button.removeEventListener('click', handleClick);

// ❌ 익명 함수는 제거 불가
button.addEventListener('click', function() {
    console.log('제거 불가');
});
button.removeEventListener('click', function() {
    // 다른 함수라서 제거 안 됨!
});

// ✅ 한 번만 실행되는 이벤트
button.addEventListener('click', function(event) {
    console.log('한 번만 실행');
}, { once: true }); // once 옵션
```

### 이벤트 객체

```javascript
button.addEventListener('click', function(event) {
    // 이벤트 정보
    console.log('타입:', event.type);              // "click"
    console.log('타겟:', event.target);            // 클릭된 요소
    console.log('현재 타겟:', event.currentTarget); // 리스너가 붙은 요소
    console.log('타임스탬프:', event.timeStamp);   // 이벤트 발생 시간

    // 마우스 정보
    console.log('X:', event.clientX);  // 뷰포트 기준
    console.log('Y:', event.clientY);
    console.log('페이지 X:', event.pageX);  // 문서 기준
    console.log('페이지 Y:', event.pageY);

    // 키보드 수정자
    console.log('Ctrl:', event.ctrlKey);
    console.log('Shift:', event.shiftKey);
    console.log('Alt:', event.altKey);
    console.log('Meta:', event.metaKey); // Cmd (Mac) / Win (Windows)

    // 기본 동작 방지
    event.preventDefault();

    // 전파 중지
    event.stopPropagation();
});
```

## 이벤트 타입

### 마우스 이벤트

```javascript
const element = document.querySelector('.box');

// click - 클릭
element.addEventListener('click', (e) => {
    console.log('클릭됨');
});

// dblclick - 더블 클릭
element.addEventListener('dblclick', (e) => {
    console.log('더블 클릭됨');
});

// mousedown / mouseup - 마우스 버튼 누름/뗌
element.addEventListener('mousedown', (e) => {
    console.log('버튼:', e.button); // 0: 왼쪽, 1: 중간, 2: 오른쪽
});

// mousemove - 마우스 이동
element.addEventListener('mousemove', (e) => {
    console.log(`위치: (${e.clientX}, ${e.clientY})`);
});

// mouseenter / mouseleave - 마우스 진입/이탈 (버블링 없음)
element.addEventListener('mouseenter', (e) => {
    element.style.backgroundColor = 'lightblue';
});

element.addEventListener('mouseleave', (e) => {
    element.style.backgroundColor = '';
});

// mouseover / mouseout - 마우스 오버/아웃 (버블링 있음)
element.addEventListener('mouseover', (e) => {
    console.log('오버:', e.target);
});

// contextmenu - 오른쪽 클릭
element.addEventListener('contextmenu', (e) => {
    e.preventDefault(); // 기본 메뉴 방지
    console.log('오른쪽 클릭됨');
});
```

```javascript
        console.log('위 화살표');
    }

// keyup - 키를 뗄 때
input.addEventListener('keyup', (e) => {
    console.log('키 업:', e.key);

// keypress - 문자 키를 눌렀을 때 (Deprecated)
// keydown을 사용하세요

input.addEventListener('keydown', (e) => {
```

## 이벤트 전파

### 캡처링과 버블링

```html
<div id="outer">
    <div id="middle">
        <button id="inner">클릭</button>
    </div>
</div>
```

```javascript
const outer = document.getElementById('outer');
const middle = document.getElementById('middle');
const inner = document.getElementById('inner');

// 버블링 (기본): inner → middle → outer
outer.addEventListener('click', () => console.log('outer'));
middle.addEventListener('click', () => console.log('middle'));
inner.addEventListener('click', () => console.log('inner'));

// 클릭 시: inner, middle, outer

// 캡처링: outer → middle → inner
outer.addEventListener('click', () => console.log('outer (캡처)'), true);
middle.addEventListener('click', () => console.log('middle (캡처)'), true);
inner.addEventListener('click', () => console.log('inner (캡처)'), true);

// 클릭 시: outer (캡처), middle (캡처), inner (캡처), inner, middle, outer

// 전파 중지
inner.addEventListener('click', (e) => {
    console.log('inner');
    e.stopPropagation(); // 여기서 중지
});
// 클릭 시: inner만 출력

// 즉시 전파 중지 (같은 요소의 다른 리스너도 실행 안 함)
inner.addEventListener('click', (e) => {
    console.log('첫 번째');
    e.stopImmediatePropagation();
});

inner.addEventListener('click', (e) => {
    console.log('두 번째'); // 실행 안 됨
});
```

### target vs currentTarget

```javascript
outer.addEventListener('click', (e) => {
    console.log('target:', e.target.id);         // 실제 클릭된 요소
    console.log('currentTarget:', e.currentTarget.id); // 리스너가 붙은 요소
});

// inner 클릭 시:
// target: inner (실제 클릭한 요소)
// currentTarget: outer (리스너가 붙은 요소)
```

```javascript
// closest를 사용한 위임

    const item = e.target.closest('.item');

        console.log('클릭됨:', item.textContent);
    }
});
```

### 실전 예제: 동적 Todo 리스트

```javascript
class TodoList {
    constructor(element) {
        this.element = element;
        this.todos = [];
        this.init();
    }

    init() {
        // 이벤트 위임: 부모 요소에 리스너 등록
        this.element.addEventListener('click', (e) => {
            // 삭제 버튼 클릭
            if (e.target.classList.contains('delete-btn')) {
                const id = e.target.dataset.id;
                this.deleteTodo(id);
            }

            // 체크박스 클릭
            if (e.target.classList.contains('todo-checkbox')) {
                const id = e.target.dataset.id;
                this.toggleTodo(id);
            }
        });

        // 폼 제출
        const form = this.element.querySelector('.todo-form');
        form.addEventListener('submit', (e) => {
            e.preventDefault();
            this.addTodo();
        });
    }

    addTodo() {
        const input = this.element.querySelector('.todo-input');
        const text = input.value.trim();

        if (!text) return;

        const todo = {
            id: Date.now().toString(),
            text,
            completed: false
        };

        this.todos.push(todo);
        this.render();
        input.value = '';
    }

    toggleTodo(id) {
        const todo = this.todos.find(t => t.id === id);
        if (todo) {
            todo.completed = !todo.completed;
            this.render();
        }
    }

    deleteTodo(id) {
        this.todos = this.todos.filter(t => t.id !== id);
        this.render();
    }

    render() {
        const list = this.element.querySelector('.todo-list');
        list.innerHTML = this.todos.map(todo => `
            <li class="todo-item ${todo.completed ? 'completed' : ''}">
                <input
                    type="checkbox"
                    class="todo-checkbox"
                    data-id="${todo.id}"
                    ${todo.completed ? 'checked' : ''}
                >
                <span>${todo.text}</span>
                <button class="delete-btn" data-id="${todo.id}">삭제</button>
            </li>
        `).join('');
    }
}
```

```javascript
    detail: { message: 'Hello' },
    bubbles: true,
    cancelable: true

// 실전 예제: 컴포넌트 간 통신
class Counter {
    constructor(element) {
        this.element = element;
        this.count = 0;
        this.init();
    }

    init() {
        const button = this.element.querySelector('button');
        button.addEventListener('click', () => {
            this.count++;
            this.notify();
        });
    }

    notify() {
        // 커스텀 이벤트 발생
        const event = new CustomEvent('countChange', {
            detail: { count: this.count },
            bubbles: true
        });
        this.element.dispatchEvent(event);
    }
}

// 다른 컴포넌트에서 수신
document.addEventListener('countChange', (e) => {
    console.log('카운트 변경됨:', e.detail.count);
});
```

### 1. 디바운싱과 쓰로틀링

```javascript
// 디바운싱: 마지막 호출 후 일정 시간 대기
function debounce(func, delay) {
    let timeout;
    return function(...args) {
        clearTimeout(timeout);
        timeout = setTimeout(() => func.apply(this, args), delay);
    };
}

// 검색 입력 디바운싱
const searchInput = document.querySelector('.search');
const search = debounce((query) => {
    console.log('검색:', query);
}, 500);

searchInput.addEventListener('input', (e) => {
    search(e.target.value);
});

// 쓰로틀링: 일정 시간 간격으로 실행
function throttle(func, limit) {
    let inThrottle;
    return function(...args) {
        if (!inThrottle) {
            func.apply(this, args);
            inThrottle = true;
            setTimeout(() => inThrottle = false, limit);
        }
    };
}

// 스크롤 쓰로틀링
const onScroll = throttle(() => {
    console.log('스크롤 Y:', window.scrollY);
}, 200);

window.addEventListener('scroll', onScroll);
```

### 2. 한 번만 실행

```javascript
// once 옵션 사용
button.addEventListener('click', () => {
    console.log('한 번만 실행');
}, { once: true });

// 직접 구현
function once(func) {
    let called = false;
    return function(...args) {
        if (!called) {
            called = true;
            return func.apply(this, args);
        }
    };
}

const initOnce = once(() => {
    console.log('초기화');
});

initOnce(); // "초기화"
initOnce(); // 실행 안 됨
```

### 3. 이벤트 버스

```javascript
class EventBus {
    constructor() {
        this.events = {};
    }

    on(event, callback) {
        if (!this.events[event]) {
            this.events[event] = [];
        }
        this.events[event].push(callback);
    }

    off(event, callback) {
        if (!this.events[event]) return;

        this.events[event] = this.events[event]
            .filter(cb => cb !== callback);
    }

    emit(event, data) {
        if (!this.events[event]) return;

        this.events[event].forEach(callback => {
            callback(data);
        });
    }
}

// 사용
const bus = new EventBus();

bus.on('userLogin', (user) => {
    console.log('사용자 로그인:', user.name);
});

bus.on('userLogin', (user) => {
    console.log('환영합니다,', user.name);
});

bus.emit('userLogin', { name: '김철수' });
// "사용자 로그인: 김철수"
// "환영합니다, 김철수"
```
