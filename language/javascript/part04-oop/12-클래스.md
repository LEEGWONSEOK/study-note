## 12.2 클래스 기본 문법

### 클래스 선언

```javascript
class Person {
    // 생성자
    constructor(name, age) {
        this.name = name;
        this.age = age;
    }

    // 메서드 (프로토타입에 추가됨)
    greet() {
        console.log(`안녕하세요, ${this.name}입니다.`);
    }

    getAge() {
        return this.age;
    }

    // Getter
    get info() {
        return `${this.name} (${this.age}세)`;
    }

    // Setter
    set info(value) {
        const [name, age] = value.split(',');
        this.name = name.trim();
        this.age = parseInt(age);
    }
}

// 사용
const person = new Person('홍길동', 25);
person.greet();  // "안녕하세요, 홍길동입니다."
console.log(person.info);  // "홍길동 (25세)"
person.info = '김철수, 30';
console.log(person.info);  // "김철수 (30세)"
```

```javascript
class User {
    constructor(name, email) {
        this.name = name;
        this.email = email;
    }

    // 정적 팩토리 메서드
    static fromJSON(json) {
```

## 12.4 상속 (Inheritance)

### extends 키워드

```javascript
class Animal {
    constructor(name) {
        this.name = name;
    }

    eat() {
        console.log(`${this.name}이(가) 먹습니다.`);
    }

    sleep() {
        console.log(`${this.name}이(가) 잡니다.`);
    }
}

class Dog extends Animal {
    constructor(name, breed) {
        super(name);  // 부모 생성자 호출 (필수!)
        this.breed = breed;
    }

    bark() {
        console.log(`${this.name}: 멍멍!`);
    }

    // 메서드 오버라이딩
    eat() {
        console.log(`${this.name}이(가) 사료를 먹습니다.`);
    }
}

const dog = new Dog('바둑이', '진돗개');
dog.eat();    // "바둑이이(가) 사료를 먹습니다."
dog.sleep();  // "바둑이이(가) 잡니다." (부모 메서드)
dog.bark();   // "바둑이: 멍멍!"
```

### 다중 상속 (Mixin 패턴)

```javascript
// Mixin 함수
const canEat = {
    eat() {
        console.log(`${this.name}이(가) 먹습니다.`);
    }
};

const canWalk = {
    walk() {
        console.log(`${this.name}이(가) 걷습니다.`);
    }
};

const canSwim = {
    swim() {
        console.log(`${this.name}이(가) 수영합니다.`);
    }
};

// Mixin 적용 함수
function mixin(target, ...sources) {
    Object.assign(target.prototype, ...sources);
}

// 사용
class Person {
    constructor(name) {
        this.name = name;
    }
}

mixin(Person, canEat, canWalk);

const person = new Person('홍길동');
person.eat();   // "홍길동이(가) 먹습니다."
person.walk();  // "홍길동이(가) 걷습니다."

// Duck은 수영도 가능
class Duck {
    constructor(name) {
        this.name = name;
    }
}

mixin(Duck, canEat, canWalk, canSwim);

const duck = new Duck('도날드');
duck.eat();
duck.walk();
duck.swim();
```

## 12.5 Private 필드와 메서드 (ES2022)

### Private 필드

```javascript
class BankAccount {
    // Private 필드 (# 접두사)
    #balance = 0;
    #pin;

    constructor(initialBalance, pin) {
        this.#balance = initialBalance;
        this.#pin = pin;
    }

    // Public 메서드
    deposit(amount) {
        if (amount > 0) {
            this.#balance += amount;
            return true;
        }
        return false;
    }

    withdraw(amount, pin) {
        if (pin !== this.#pin) {
            throw new Error('잘못된 PIN');
        }

        if (amount > this.#balance) {
            throw new Error('잔액 부족');
        }

        this.#balance -= amount;
        return amount;
    }

    getBalance(pin) {
        if (pin !== this.#pin) {
            throw new Error('잘못된 PIN');
        }
        return this.#balance;
    }

    // Private 메서드
    #validatePin(pin) {
        return pin === this.#pin;
    }
}

const account = new BankAccount(1000, '1234');
account.deposit(500);
console.log(account.getBalance('1234'));  // 1500

// ❌ Private 필드에 직접 접근 불가
console.log(account.#balance);  // SyntaxError!
```

**Java와 비교**:

## 12.6 Getter와 Setter

```javascript
class Temperature {
    constructor(celsius) {
        this._celsius = celsius;
    }

    // Getter
    get celsius() {
        return this._celsius;
    }

    get fahrenheit() {
        return this._celsius * 9/5 + 32;
    }

    get kelvin() {
        return this._celsius + 273.15;
    }

    // Setter
    set celsius(value) {
        if (value < -273.15) {
            throw new Error('절대온도 이하');
        }
        this._celsius = value;
    }

    set fahrenheit(value) {
        this.celsius = (value - 32) * 5/9;
    }
}

const temp = new Temperature(25);
console.log(temp.celsius);     // 25
console.log(temp.fahrenheit);  // 77
console.log(temp.kelvin);      // 298.15

temp.fahrenheit = 100;
console.log(temp.celsius);  // 37.78
```

## 12.7 실전 예제

### 유효성 검사 클래스

```javascript
class Validator {
    static isEmail(email) {
        const pattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        return pattern.test(email);
    }

    static isURL(url) {
        try {
            new URL(url);
            return true;
        } catch {
            return false;
        }
    }

    static isPhone(phone) {
        const pattern = /^01[0-9]-?[0-9]{4}-?[0-9]{4}$/;
        return pattern.test(phone);
    }

    static min(value, min) {
        return value >= min;
    }

    static max(value, max) {
        return value <= max;
    }

    static minLength(str, length) {
        return str.length >= length;
    }
}

// 사용
console.log(Validator.isEmail('hong@example.com'));  // true
console.log(Validator.isEmail('invalid-email'));  // false
console.log(Validator.isPhone('010-1234-5678'));  // true
```

### API 클라이언트 클래스

```javascript
class ApiClient {
    #baseURL;
    #headers;

    constructor(baseURL, headers = {}) {
        this.#baseURL = baseURL;
        this.#headers = {
            'Content-Type': 'application/json',
            ...headers
        };
    }

    async #request(endpoint, options = {}) {
        const url = `${this.#baseURL}${endpoint}`;
        const config = {
            ...options,
            headers: {
                ...this.#headers,
                ...options.headers
            }
        };

        const response = await fetch(url, config);

        if (!response.ok) {
            throw new Error(`HTTP ${response.status}`);
        }

        return response.json();
    }

    async get(endpoint) {
        return this.#request(endpoint);
    }

    async post(endpoint, data) {
        return this.#request(endpoint, {
            method: 'POST',
            body: JSON.stringify(data)
        });
    }

    async put(endpoint, data) {
        return this.#request(endpoint, {
            method: 'PUT',
            body: JSON.stringify(data)
        });
    }

    async delete(endpoint) {
        return this.#request(endpoint, {
            method: 'DELETE'
        });
    }

    setHeader(key, value) {
        this.#headers[key] = value;
    }
}

// 사용
const api = new ApiClient('https://api.example.com');
api.setHeader('Authorization', 'Bearer token123');

const users = await api.get('/users');
const newUser = await api.post('/users', { name: '홍길동' });
```

### 상태 관리 클래스

```javascript
class Store {
    #state;
    #listeners = [];

    constructor(initialState = {}) {
        this.#state = initialState;
    }

    getState() {
        return { ...this.#state };  // 복사본 반환
    }

    setState(updates) {
        const prevState = this.#state;
        this.#state = {
            ...this.#state,
            ...updates
        };

        this.#notify(this.#state, prevState);
    }

    subscribe(listener) {
        this.#listeners.push(listener);

        // 구독 취소 함수 반환
        return () => {
            this.#listeners = this.#listeners.filter(l => l !== listener);
        };
    }

    #notify(newState, prevState) {
        this.#listeners.forEach(listener => {
            listener(newState, prevState);
        });
    }
}

// 사용
const store = new Store({ count: 0, user: null });

const unsubscribe = store.subscribe((newState, prevState) => {
    console.log('상태 변경:', prevState, '→', newState);
});

store.setState({ count: 1 });
// "상태 변경: { count: 0, user: null } → { count: 1, user: null }"

store.setState({ user: { name: '홍길동' } });
// "상태 변경: { count: 1, user: null } → { count: 1, user: { name: '홍길동' } }"

unsubscribe();
store.setState({ count: 2 });  // 출력 없음 (구독 취소됨)
```

## 📝 핵심 요약

1. **ES6 클래스**: 프로토타입의 문법 설탕
   - constructor: 생성자
   - 메서드: 프로토타입에 추가
   - static: 정적 메서드/속성

2. **상속**: extends와 super
   - 부모 클래스 상속

   - Private 필드/메서드
   - 클래스 외부에서 접근 불가

4. **Getter/Setter**: 계산된 속성
   - get: 속성처럼 접근
   - set: 속성처럼 할당

```javascript
// 1. Stack 클래스
class Stack {
    #items = [];

    push(element) {
        this.#items.push(element);
    }

    pop() {
        if (this.isEmpty()) {
            throw new Error('Stack is empty');
        }
        return this.#items.pop();
    }

    peek() {
        if (this.isEmpty()) {
            return undefined;
        }
        return this.#items[this.#items.length - 1];
    }

    isEmpty() {
        return this.#items.length === 0;
    }

    get size() {
        return this.#items.length;
    }

    clear() {
        this.#items = [];
    }
}

class Counter {
    #count = 0;
    #listeners = [];

    get value() {
        return this.#count;
    }

    increment() {
        this.#count++;
        this.#notify();
    }

    decrement() {
        this.#count--;
        this.#notify();
    }

    reset() {
        this.#count = 0;
        this.#notify();
    }

    subscribe(listener) {
        this.#listeners.push(listener);
        return () => {
            this.#listeners = this.#listeners.filter(l => l !== listener);
        };
    }

    #notify() {
        this.#listeners.forEach(listener => {
            listener(this.#count);
        });
    }
}

// 사용
```
