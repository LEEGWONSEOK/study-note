# Chapter 14. 함수형 개념

## 14.1 함수형 프로그래밍이란?

> 🎯 **핵심**: 함수형 프로그래밍은 **순수 함수와 불변성**을 중심으로 하는 패러다임입니다.
> - 부수 효과(Side Effect) 최소화
> - 데이터 불변성
> - 함수 조합

**비유**: 함수형 프로그래밍은 **레고 블록**입니다.
- 각 블록(함수)은 독립적
- 블록을 조합해서 복잡한 구조 생성
- 같은 입력에 항상 같은 결과

**Java Stream API와 비교**:
```java
// Java Stream API (함수형 스타일)
List<Integer> numbers = Arrays.asList(1, 2, 3, 4, 5);
int sum = numbers.stream()
    .filter(n -> n % 2 == 0)
    .map(n -> n * 2)
    .reduce(0, (a, b) -> a + b);

// JavaScript (더 간결)
const numbers = [1, 2, 3, 4, 5];
const sum = numbers
    .filter(n => n % 2 === 0)
    .map(n => n * 2)
    .reduce((a, b) => a + b, 0);
```

## 14.2 순수 함수 (Pure Function)

### 정의

**순수 함수 조건**:
1. 같은 입력 → 항상 같은 출력
2. 부수 효과(Side Effect) 없음

```javascript
// ✅ 순수 함수
function add(a, b) {
    return a + b;
}

console.log(add(2, 3));  // 항상 5

// ✅ 순수 함수
function multiply(arr, factor) {
    return arr.map(n => n * factor);
}

const numbers = [1, 2, 3];
console.log(multiply(numbers, 2));  // [2, 4, 6]
console.log(numbers);  // [1, 2, 3] (원본 불변)

// ❌ 비순수 함수 - 외부 상태 의존
let count = 0;

function increment() {
    count++;  // 외부 상태 변경!
    return count;
}

console.log(increment());  // 1
console.log(increment());  // 2 (같은 입력에 다른 출력!)

// ❌ 비순수 함수 - 부수 효과
function addToArray(arr, value) {
    arr.push(value);  // 원본 수정!
    return arr;
}

const nums = [1, 2];
addToArray(nums, 3);
console.log(nums);  // [1, 2, 3] (원본 변경됨!)
```

### 순수 함수의 장점

```javascript
// 1. 테스트 용이
function calculateTax(price, rate) {
    return price * rate;
}

// 테스트하기 쉬움
console.assert(calculateTax(100, 0.1) === 10);
console.assert(calculateTax(200, 0.15) === 30);

// 2. 예측 가능
const result1 = add(2, 3);
const result2 = add(2, 3);
console.log(result1 === result2);  // true (항상 같은 결과)

// 3. 병렬 처리 안전
// 순수 함수는 상태를 공유하지 않으므로 병렬 처리 안전

// 4. 메모이제이션 가능
function memoize(fn) {
    const cache = {};
    return function(...args) {
        const key = JSON.stringify(args);
        if (key in cache) {
            return cache[key];
        }
        cache[key] = fn(...args);
        return cache[key];
    };
}

const expensiveOperation = memoize((n) => {
    console.log('계산 중...');
    return n * n;
});

console.log(expensiveOperation(5));  // "계산 중...", 25
console.log(expensiveOperation(5));  // 25 (캐시 사용)
```

## 14.3 불변성 (Immutability)

### 개념

```javascript
// ❌ 가변 (Mutable)
const person = { name: '홍길동', age: 25 };
person.age = 26;  // 원본 수정
console.log(person);  // { name: '홍길동', age: 26 }

// ✅ 불변 (Immutable)
const person = { name: '홍길동', age: 25 };
const updated = { ...person, age: 26 };  // 새 객체 생성
console.log(person);   // { name: '홍길동', age: 25 } (원본 유지)
console.log(updated);  // { name: '홍길동', age: 26 }
```

## 14.4 고차 함수 (Higher-Order Function)

### 정의

**고차 함수**: 함수를 인자로 받거나 함수를 반환하는 함수

```javascript
// 1. 함수를 인자로 받는 함수
function repeat(n, action) {
    for (let i = 0; i < n; i++) {
        action(i);
    }
}

repeat(3, console.log);  // 0, 1, 2

// 2. 함수를 반환하는 함수
function multiplier(factor) {
    return function(number) {
        return number * factor;
    };
}

const double = multiplier(2);
const triple = multiplier(3);

console.log(double(5));  // 10
console.log(triple(5));  // 15
```

---

### 내장 고차 함수

```javascript
const numbers = [1, 2, 3, 4, 5];

// map - 변환
const doubled = numbers.map(n => n * 2);
// [2, 4, 6, 8, 10]

// filter - 필터링
const evens = numbers.filter(n => n % 2 === 0);
// [2, 4]

// reduce - 축약
const sum = numbers.reduce((acc, n) => acc + n, 0);
// 15

// forEach - 반복 (반환값 없음)
numbers.forEach(n => console.log(n));

// find - 찾기
const found = numbers.find(n => n > 3);
// 4

// some - 하나라도 만족?
const hasEven = numbers.some(n => n % 2 === 0);
// true

// every - 모두 만족?
const allPositive = numbers.every(n => n > 0);
// true
```

### 고차 함수 만들기

```javascript
// 1. 조건 반전
function not(predicate) {
    return function(...args) {
        return !predicate(...args);
    };
}

const isEven = n => n % 2 === 0;
const isOdd = not(isEven);

console.log(isEven(2));  // true
console.log(isOdd(2));   // false

// 2. 함수 조합
function compose(...fns) {
    return function(value) {
        return fns.reduceRight((acc, fn) => fn(acc), value);
    };
}

const add10 = n => n + 10;
const multiply2 = n => n * 2;
const subtract5 = n => n - 5;

const combined = compose(subtract5, multiply2, add10);
console.log(combined(5));  // (5 + 10) * 2 - 5 = 25

// 3. Partial Application
function partial(fn, ...presetArgs) {
    return function(...laterArgs) {
        return fn(...presetArgs, ...laterArgs);
    };
}

function greet(greeting, name) {
    return `${greeting}, ${name}!`;
}

const greetHello = partial(greet, '안녕하세요');
console.log(greetHello('홍길동'));  // "안녕하세요, 홍길동!"
```

## 14.5 커링 (Currying)

### 정의

**커링**: 여러 인자를 받는 함수를 단일 인자를 받는 함수들의 체인으로 변환

```javascript
// 일반 함수
function add(a, b, c) {
    return a + b + c;
}

console.log(add(1, 2, 3));  // 6

// 커링된 함수
function curriedAdd(a) {
    return function(b) {
        return function(c) {
            return a + b + c;
        };
    };
}

console.log(curriedAdd(1)(2)(3));  // 6

// 화살표 함수로 더 간결하게
const curriedAdd = a => b => c => a + b + c;

console.log(curriedAdd(1)(2)(3));  // 6
```

### 커링 유틸리티

```javascript
// 범용 커링 함수
function curry(fn) {
    return function curried(...args) {
        if (args.length >= fn.length) {
            return fn.apply(this, args);
        } else {
            return function(...moreArgs) {
                return curried.apply(this, args.concat(moreArgs));
            };
        }
    };
}

// 사용
function sum(a, b, c) {
    return a + b + c;
}

const curriedSum = curry(sum);

console.log(curriedSum(1)(2)(3));      // 6
console.log(curriedSum(1, 2)(3));      // 6
console.log(curriedSum(1)(2, 3));      // 6
console.log(curriedSum(1, 2, 3));      // 6
```

## 14.6 함수 조합 (Function Composition)

### pipe와 compose

```javascript
// compose - 오른쪽에서 왼쪽으로
const compose = (...fns) => value =>
    fns.reduceRight((acc, fn) => fn(acc), value);

// pipe - 왼쪽에서 오른쪽으로
const pipe = (...fns) => value =>
    fns.reduce((acc, fn) => fn(acc), value);

// 함수들
const add10 = n => n + 10;
const multiply2 = n => n * 2;
const subtract5 = n => n - 5;

// compose 사용
const composedFn = compose(subtract5, multiply2, add10);
console.log(composedFn(5));  // (5 + 10) * 2 - 5 = 25

// pipe 사용 (더 직관적)
const pipedFn = pipe(add10, multiply2, subtract5);
console.log(pipedFn(5));  // (5 + 10) * 2 - 5 = 25
```

---

### 실용 예제

```javascript
// 사용자 데이터 처리
const users = [
    { id: 1, name: '홍길동', age: 25, active: true },
    { id: 2, name: '김철수', age: 30, active: false },
    { id: 3, name: '이영희', age: 28, active: true }
];

// 작은 함수들
const filterActive = users => users.filter(u => u.active);
const sortByAge = users => [...users].sort((a, b) => a.age - b.age);
const mapNames = users => users.map(u => u.name);

// 조합
const getActiveSortedNames = pipe(
    filterActive,
    sortByAge,
    mapNames
);

console.log(getActiveSortedNames(users));
// ['홍길동', '이영희']
```

## 14.7 Functor와 Monad 개념

### Functor

```javascript
// Functor: map 메서드를 가진 컨테이너
class Box {
    constructor(value) {
        this._value = value;
    }

    map(fn) {
        return new Box(fn(this._value));
    }

    fold(fn) {
        return fn(this._value);
    }
}

// 사용
const result = new Box(5)
    .map(n => n * 2)    // 10
    .map(n => n + 3)    // 13
    .fold(n => n);      // 13

console.log(result);  // 13
```

---

### Maybe (Option) Monad

```javascript
class Maybe {
    constructor(value) {
        this._value = value;
    }

    static of(value) {
        return new Maybe(value);
    }

    isNothing() {
        return this._value === null || this._value === undefined;
    }

    map(fn) {
        return this.isNothing()
            ? Maybe.of(null)
            : Maybe.of(fn(this._value));
    }

    getOrElse(defaultValue) {
        return this.isNothing() ? defaultValue : this._value;
    }
}

// 사용
const user = { name: '홍길동', address: { city: '서울' } };

// ❌ 일반 방식 - null 체크 지옥
function getCity(user) {
    if (user && user.address && user.address.city) {
        return user.address.city.toUpperCase();
    }
    return 'UNKNOWN';
}

// ✅ Maybe 사용
const getCity2 = user =>
    Maybe.of(user)
        .map(u => u.address)
        .map(a => a.city)
        .map(c => c.toUpperCase())
        .getOrElse('UNKNOWN');

console.log(getCity2(user));  // "서울"
console.log(getCity2(null));  // "UNKNOWN"
```

## 📝 핵심 요약

1. **순수 함수**: 같은 입력 → 같은 출력, 부수 효과 없음
   - 테스트 용이, 예측 가능
   - 메모이제이션 가능

2. **불변성**: 데이터 수정 대신 새 데이터 생성
   - 스프레드 연산자 활용
   - 원본 보존

3. **고차 함수**: 함수를 다루는 함수
   - map, filter, reduce
   - 함수 조합, 커링

4. **커링**: 다중 인자 → 단일 인자 체인
   - 부분 적용
   - 재사용성 향상

5. **함수 조합**: 작은 함수 → 큰 함수
   - pipe, compose
   - 가독성, 재사용성
