## React 이벤트 시스템

React는 **합성 이벤트(SyntheticEvent)** 시스템을 사용합니다.

### 비유로 이해하기
React 이벤트는 **통역사**와 같습니다:
- 브라우저마다 다른 이벤트 방식을 표준화
- 일관된 인터페이스 제공
- 크로스 브라우저 호환성 보장
- 성능 최적화 자동 처리

function KeyboardEvents() {
    const [key, setKey] = useState('');
    const [text, setText] = useState('');

    const handleKeyDown = (e) => {
        setKey(e.key);

        // 특정 키 조합

            e.preventDefault();
            console.log('Ctrl+S 저장');
        }

        if (e.key === 'Enter') {
            console.log('엔터 키');
        }

        if (e.key === 'Escape') {
            setText('');
        }
    };

    const handleKeyPress = (e) => {
        // 숫자만 입력 허용

            e.preventDefault();
        }
    };

    return (
        <div>
            <input
                value={text}
                onChange={(e) => setText(e.target.value)}
                onKeyDown={handleKeyDown}
                placeholder="텍스트 입력 (ESC로 초기화)"
            />
            <p>마지막 키: {key}</p>

            <input
                onKeyPress={handleKeyPress}
                placeholder="숫자만 입력 가능"
            />
        </div>

## 제어 컴포넌트 vs 비제어 컴포넌트

### 제어 컴포넌트 (Controlled)

React state가 **유일한 진실 공급원(single source of truth)**입니다.

```javascript
function ControlledInput() {
    const [value, setValue] = useState('');

    const handleChange = (e) => {
        // 입력값을 완전히 제어
        let newValue = e.target.value;

        // 예: 대문자로만 입력
        newValue = newValue.toUpperCase();

        // 예: 최대 10글자
        if (newValue.length <= 10) {
            setValue(newValue);
        }
    };

    return (
        <div>
            {/* React state와 동기화됨 */}
            <input
                value={value}
                onChange={handleChange}
                placeholder="대문자만 (최대 10자)"
            />
            <p>입력값: {value}</p>
        </div>
    );
}

// 제어 컴포넌트의 장점:
// - 실시간 유효성 검사
// - 입력값 변환/제한
// - 조건부 버튼 활성화
// - 즉시 피드백 제공
```

### 비제어 컴포넌트 (Uncontrolled)

DOM이 **진실 공급원**이며, ref로 접근합니다.

```javascript
function UncontrolledInput() {
    const inputRef = useRef(null);

        <form onSubmit={handleSubmit}>
            {/* React state 없이 DOM이 관리 */}
            <input

                defaultValue="초기값"
                placeholder="비제어 입력"
            />
            <button type="submit">제출</button>
        </form>
    );
}

// 비제어 컴포넌트의 장점:
// - 간단한 구현
// - 불필요한 리렌더링 없음
// - 파일 입력 등에 유용
```

### 1. 검증 기능이 있는 폼

```javascript
function ValidationForm() {
    const [formData, setFormData] = useState({
        username: '',
        email: '',
        password: '',
        confirmPassword: ''
    });

    const [errors, setErrors] = useState({});
    const [touched, setTouched] = useState({});

    // 실시간 유효성 검사
    const validate = (name, value) => {
        let error = '';

        switch (name) {
            case 'username':
                if (!value) {
                    error = '사용자명을 입력하세요';
                } else if (value.length < 3) {
                    error = '3자 이상 입력하세요';
                }
                break;

            case 'email':
                if (!value) {
                    error = '이메일을 입력하세요';
                } else if (!/\S+@\S+\.\S+/.test(value)) {
                    error = '유효한 이메일을 입력하세요';
                }
                break;

            case 'password':
                if (!value) {
                    error = '비밀번호를 입력하세요';
                } else if (value.length < 8) {
                    error = '8자 이상 입력하세요';
                }
                break;

            case 'confirmPassword':
                if (value !== formData.password) {
                    error = '비밀번호가 일치하지 않습니다';
                }
                break;
        }

        return error;
    };

    const handleChange = (e) => {
        const { name, value } = e.target;

        setFormData(prev => ({
            ...prev,
            [name]: value
        }));

        // 이미 터치된 필드만 실시간 검사
        if (touched[name]) {
            const error = validate(name, value);
            setErrors(prev => ({
                ...prev,
                [name]: error
            }));
        }
    };

    const handleBlur = (e) => {
        const { name, value } = e.target;

        setTouched(prev => ({
            ...prev,
            [name]: true
        }));

        const error = validate(name, value);
        setErrors(prev => ({
            ...prev,
            [name]: error
        }));
    };

    const handleSubmit = (e) => {
        e.preventDefault();

        // 모든 필드 검사
        const newErrors = {};
        Object.keys(formData).forEach(key => {
            const error = validate(key, formData[key]);
            if (error) newErrors[key] = error;
        });

        setErrors(newErrors);
        setTouched(
            Object.keys(formData).reduce((acc, key) => ({ ...acc, [key]: true }), {})
        );

        if (Object.keys(newErrors).length === 0) {
            console.log('폼 제출:', formData);
        }
    };

    return (
        <form onSubmit={handleSubmit}>
            <div>
                <input
                    name="username"
                    value={formData.username}
                    onChange={handleChange}
                    onBlur={handleBlur}
                    placeholder="사용자명"
                />
                {touched.username && errors.username && (
                    <span className="error">{errors.username}</span>
                )}
            </div>

            <div>
                <input
                    name="email"
                    type="email"
                    value={formData.email}
                    onChange={handleChange}
                    onBlur={handleBlur}
                    placeholder="이메일"
                />
                {touched.email && errors.email && (
                    <span className="error">{errors.email}</span>
                )}
            </div>

            <div>
                <input
                    name="password"
                    type="password"
                    value={formData.password}
                    onChange={handleChange}
                    onBlur={handleBlur}
                    placeholder="비밀번호"
                />
                {touched.password && errors.password && (
                    <span className="error">{errors.password}</span>
                )}
            </div>

            <div>
                <input
                    name="confirmPassword"
                    type="password"
                    value={formData.confirmPassword}
                    onChange={handleChange}
                    onBlur={handleBlur}
                    placeholder="비밀번호 확인"
                />
                {touched.confirmPassword && errors.confirmPassword && (
                    <span className="error">{errors.confirmPassword}</span>
                )}
            </div>

            <button type="submit">가입하기</button>
        </form>
    );
}

function FileUpload() {
    const [files, setFiles] = useState([]);
    const [isDragging, setIsDragging] = useState(false);

    const handleDragEnter = (e) => {
        e.preventDefault();
        setIsDragging(true);
    };

    const handleDragLeave = (e) => {
        e.preventDefault();
        setIsDragging(false);
    };

    const handleDragOver = (e) => {
        e.preventDefault();
    };

    const handleDrop = (e) => {
        e.preventDefault();
        setIsDragging(false);

        const droppedFiles = Array.from(e.dataTransfer.files);
        handleFiles(droppedFiles);
    };

    const handleFileInput = (e) => {
        const selectedFiles = Array.from(e.target.files);
        handleFiles(selectedFiles);
    };

    const handleFiles = (newFiles) => {
        // 이미지만 허용
        const imageFiles = newFiles.filter(file =>
            file.type.startsWith('image/')
        );

        setFiles(prev => [...prev, ...imageFiles]);
    };

    const removeFile = (index) => {
        setFiles(prev => prev.filter((_, i) => i !== index));
    };

    return (
        <div>
            <div
                className={`dropzone ${isDragging ? 'dragging' : ''}`}
                onDragEnter={handleDragEnter}
                onDragLeave={handleDragLeave}
                onDragOver={handleDragOver}
                onDrop={handleDrop}
                style={{
                    border: '2px dashed #ccc',
                    padding: '50px',
                    textAlign: 'center',
                    background: isDragging ? '#eee' : '#fff'
                }}
            >
                <p>파일을 드래그하거나 클릭하여 선택하세요</p>
                <input
                    type="file"
                    multiple
                    accept="image/*"
                    onChange={handleFileInput}
                    style={{ display: 'none' }}
                    id="file-input"
                />
                <label htmlFor="file-input">
                    <button type="button" onClick={() =>
                        document.getElementById('file-input').click()
                    }>
                        파일 선택
                    </button>
                </label>
            </div>

            {files.length > 0 && (
                <div>
                    <h3>선택된 파일 ({files.length}개)</h3>
                    <ul>
                        {files.map((file, index) => (
                            <li key={index}>
                                {file.name} ({(file.size / 1024).toFixed(2)} KB)
                                <button onClick={() => removeFile(index)}>
                                    삭제
                                </button>
                            </li>
                        ))}
                    </ul>
                </div>
            )}
        </div>
    );
}

function Autocomplete() {
    const [query, setQuery] = useState('');
    const [suggestions, setSuggestions] = useState([]);
    const [selectedIndex, setSelectedIndex] = useState(-1);
    const [isOpen, setIsOpen] = useState(false);

    const items = [
        '사과', '바나나', '오렌지', '포도', '딸기',
        '수박', '참외', '메론', '복숭아', '자두'
    ];

    useEffect(() => {
        if (query.length > 0) {
            const filtered = items.filter(item =>
                item.includes(query)
            );
            setSuggestions(filtered);
            setIsOpen(filtered.length > 0);
        } else {
            setSuggestions([]);
            setIsOpen(false);
        }
        setSelectedIndex(-1);
    }, [query]);

    const handleKeyDown = (e) => {
        if (!isOpen) return;

        switch (e.key) {
            case 'ArrowDown':
                e.preventDefault();
                setSelectedIndex(prev =>
                    prev < suggestions.length - 1 ? prev + 1 : prev
                );
                break;

            case 'ArrowUp':
                e.preventDefault();
                setSelectedIndex(prev => prev > 0 ? prev - 1 : -1);
                break;

            case 'Enter':
                e.preventDefault();
                if (selectedIndex >= 0) {
                    selectItem(suggestions[selectedIndex]);
                }
                break;

            case 'Escape':
                setIsOpen(false);
                break;
        }
    };

    const selectItem = (item) => {
        setQuery(item);
        setIsOpen(false);
        setSelectedIndex(-1);
    };

    return (
        <div className="autocomplete">
            <input
                value={query}
                onChange={(e) => setQuery(e.target.value)}
                onKeyDown={handleKeyDown}
                onFocus={() => query && setIsOpen(true)}
                onBlur={() => setTimeout(() => setIsOpen(false), 200)}
                placeholder="과일 검색..."
            />

            {isOpen && (
                <ul className="suggestions">
                    {suggestions.map((item, index) => (
                        <li
                            key={item}
                            className={index === selectedIndex ? 'selected' : ''}
                            onClick={() => selectItem(item)}
                            onMouseEnter={() => setSelectedIndex(index)}
                        >
                            {item}
                        </li>
                    ))}
                </ul>
            )}
        </div>
    );
}
```

function SearchWithDebounce() {
    const [query, setQuery] = useState('');
    const [results, setResults] = useState([]);

    useEffect(() => {
        // 입력 후 500ms 대기
        const timer = setTimeout(() => {
            if (query) {
                // API 호출
                fetch(`/api/search?q=${query}`)
                    .then(res => res.json())
                    .then(setResults);
            } else {
                setResults([]);
            }
        }, 500);

        // Cleanup: 타이머 취소
        return () => clearTimeout(timer);
    }, [query]);

    return (
        <div>
            <input
                value={query}
                onChange={(e) => setQuery(e.target.value)}
                placeholder="검색..."
            />

    );
}

### 2. 쓰로틀링 (Throttling)

```javascript
function ScrollWithThrottle() {
    const [scrollY, setScrollY] = useState(0);

    useEffect(() => {
        let ticking = false;

        const handleScroll = () => {
            if (!ticking) {
                window.requestAnimationFrame(() => {
                    setScrollY(window.scrollY);
                    ticking = false;
                });
                ticking = true;
            }
        };

        window.addEventListener('scroll', handleScroll);
        return () => window.removeEventListener('scroll', handleScroll);
    }, []);

    return <div>스크롤 위치: {scrollY}px</div>;
}
```

### 3. useCallback으로 함수 메모이제이션

```javascript
function OptimizedList() {
    const [items, setItems] = useState([1, 2, 3, 4, 5]);

    // ❌ 매 렌더링마다 새 함수 생성
    const handleDelete = (id) => {
        setItems(prev => prev.filter(item => item !== id));
    };

    // ✅ 함수 메모이제이션
    const handleDeleteOptimized = useCallback((id) => {
        setItems(prev => prev.filter(item => item !== id));
    }, []); // 의존성 없음

    return (
        <ul>
            {items.map(item => (
                <ListItem
                    key={item}
                    id={item}
                    onDelete={handleDeleteOptimized}
                />
            ))}
        </ul>
    );
}

const ListItem = React.memo(({ id, onDelete }) => {
    console.log('렌더링:', id);
    return (
        <li>
            Item {id}
            <button onClick={() => onDelete(id)}>삭제</button>
        </li>
    );
});
```
